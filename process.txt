from num2words import num2words
import re

def number_to_words_es(value: int) -> str:
    """
    Convierte un entero a palabras en español.
    """
    return num2words(value, lang="es")


def decimal_to_words_es(value: str) -> str:
    """
    Convierte un número decimal a palabras.
    Ejemplo:
        15.5 -> quince coma cinco
    """
    value = value.replace(",", ".")

    try:
        number = Decimal(value)

        if number == number.to_integral():
            return number_to_words_es(int(number))

        integer_part = int(number)
        decimal_part = str(number).split(".")[1]

        return (
            f"{number_to_words_es(integer_part)} "
            f"coma "
            f"{' '.join(number_to_words_es(int(d)) for d in decimal_part)}"
        )

    except Exception:
        return value

def final_response_to_words(text: str) -> str:
    """
    Convierte números de la respuesta final a palabras.

    IMPORTANTE:
    Esta función solamente debe ejecutarse justo antes de devolver
    la respuesta al cliente.

    No debe utilizarse sobre:
      - prompts
      - mensajes del usuario
      - mensajes del asistente
      - argumentos de herramientas
      - resultados JSON de herramientas
    """

    if not text:
        return text

    # ---------------------------------------------------------
    # 1. Fechas ISO: YYYY-MM-DD
    # ---------------------------------------------------------

    def replace_iso_date(match):
        year = int(match.group(1))
        month = int(match.group(2))
        day = int(match.group(3))

        month_name = MONTHS_ES[month - 1]

        return (
            f"{number_to_words_es(day)} "
            f"de {month_name} "
            f"de {number_to_words_es(year)}"
        )

    text = re.sub(
        r"\b(\d{4})-(\d{2})-(\d{2})\b",
        replace_iso_date,
        text
    )

    # ---------------------------------------------------------
    # 2. Valores monetarios colombianos
    #
    # 200.000
    # 1.250.000
    # $200.000
    # ---------------------------------------------------------

    def replace_money(match):
        prefix = match.group(1) or ""
        number = match.group(2)

        clean_number = number.replace(".", "").replace(",", "")

        try:
            value = int(clean_number)
        except ValueError:
            return match.group(0)

        words = number_to_words_es(value)

        if prefix.strip() == "$":
            return f"{words} pesos"

        return f"{words} pesos"

    text = re.sub(
        r"(\$)?\s*\b(\d{1,3}(?:\.\d{3})+|\d+)\s*(?=pesos|COP\b)",
        replace_money,
        text,
        flags=re.IGNORECASE
    )

    # ---------------------------------------------------------
    # 3. Números con separador de miles que todavía no
    #    sean monetarios.
    #
    #    1.500 días -> mil quinientos días
    # ---------------------------------------------------------

    def replace_thousands(match):
        number = match.group(0)

        clean_number = number.replace(".", "").replace(",", "")

        try:
            value = int(clean_number)
        except ValueError:
            return number

        return number_to_words_es(value)

    text = re.sub(
        r"\b\d{1,3}(?:\.\d{3})+\b",
        replace_thousands,
        text
    )

    # ---------------------------------------------------------
    # 4. Números enteros simples
    #
    #    15 días -> quince días
    #    3 pagos -> tres pagos
    # ---------------------------------------------------------

    def replace_integer(match):
        number = match.group(0)

        try:
            value = int(number)
        except ValueError:
            return number

        return number_to_words_es(value)

    text = re.sub(
        r"\b\d+\b",
        replace_integer,
        text
    )

    return text




def extract_final_response(
    self,
    state: AgentState,
) -> str:

    messages = state["messages"]

    # ----------------------------------------------------
    # Primero buscamos tools terminales.
    # ----------------------------------------------------

    terminal_messages = [
        message
        for message in messages
        if (
            isinstance(message, ToolMessage)
            and message.name in TERMINAL_TOOLS
        )
    ]

    if terminal_messages:

        responses = []

        for message in terminal_messages:

            formatted = ResponseFormatter.format(
                message.name,
                message.content,
            )

            responses.append(formatted)

        # Si existe más de una herramienta terminal,
        # devolvemos cada resultado en una línea.
        response = "\n".join(responses)

        # ------------------------------------------------
        # Conversión FINAL de números a palabras.
        # Aquí ya no afecta las herramientas.
        # ------------------------------------------------
        return final_response_to_words(response)

    # ----------------------------------------------------
    # Si no hubo tools, usamos la respuesta normal del LLM.
    # ----------------------------------------------------

    for message in reversed(messages):

        if isinstance(message, AIMessage):

            content = message.content

            if isinstance(content, str):
                return final_response_to_words(content.strip())

            if isinstance(content, list):

                text_parts = []

                for item in content:

                    if isinstance(item, dict):

                        text = item.get("text")

                        if text:
                            text_parts.append(text)

                if text_parts:
                    response = "".join(text_parts).strip()

                    return final_response_to_words(response)

    return ""
