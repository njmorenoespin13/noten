#!/usr/bin/env python3

import argparse
import asyncio
import inspect
import json
import signal
import time
from dataclasses import dataclass, field
from typing import Optional

import grpc

import audio_listener_pb2
import audio_listener_pb2_grpc

# Ajusta este import al nombre real de tu clase
from worker_process import workerProcess


# ============================================================
# CONFIGURACIÓN
# ============================================================

SESSION_TIMEOUT_SECONDS = 10 * 60
CLEANUP_INTERVAL_SECONDS = 30


# ============================================================
# SESIÓN
# ============================================================

@dataclass
class CallSession:
    call_id: str

    metadata_json: str = ""

    created_at: float = field(default_factory=time.monotonic)
    last_activity: float = field(default_factory=time.monotonic)

    processor: Optional[workerProcess] = None

    processing: bool = False
    cleanup_requested: bool = False

    lock: asyncio.Lock = field(default_factory=asyncio.Lock)

    def touch(self):
        self.last_activity = time.monotonic()

    def idle_seconds(self) -> float:
        return time.monotonic() - self.last_activity


# ============================================================
# SERVICIO gRPC
# ============================================================

class AudioOutService(audio_listener_pb2_grpc.SegmentOutputServicer):

    def __init__(
        self,
        block_active: bool = False,
    ):
        self.block_active = block_active

        self.sessions: dict[str, CallSession] = {}

        # Protege creación/eliminación de sesiones
        self.sessions_lock = asyncio.Lock()

        self.cleanup_task: Optional[asyncio.Task] = None

        self.running = True

    # --------------------------------------------------------
    # START
    # --------------------------------------------------------

    async def start(self):
        self.cleanup_task = asyncio.create_task(
            self._cleanup_loop()
        )

        print(
            f"[OUT] Servicio iniciado | "
            f"block_active={self.block_active}"
        )

    # --------------------------------------------------------
    # STOP
    # --------------------------------------------------------

    async def stop(self):
        self.running = False

        if self.cleanup_task:
            self.cleanup_task.cancel()

            try:
                await self.cleanup_task
            except asyncio.CancelledError:
                pass

            self.cleanup_task = None

        async with self.sessions_lock:
            sessions = list(self.sessions.values())
            self.sessions.clear()

        print(
            f"[OUT] Cerrando {len(sessions)} sesiones..."
        )

        # Limpieza concurrente
        await asyncio.gather(
            *(self._cleanup_session(session) for session in sessions),
            return_exceptions=True,
        )

        print("[OUT] Servicio detenido")

    # --------------------------------------------------------
    # OBTENER / CREAR SESIÓN
    # --------------------------------------------------------

    async def _get_or_create_session(
        self,
        call_id: str,
    ) -> CallSession:

        async with self.sessions_lock:

            session = self.sessions.get(call_id)

            if session is not None:
                return session

            print(
                f"[OUT] Nueva sesión | call_id={call_id}"
            )

            session = CallSession(
                call_id=call_id
            )

            session.processor = workerProcess()

            self.sessions[call_id] = session

            return session

    # --------------------------------------------------------
    # METADATA
    # --------------------------------------------------------

    @staticmethod
    def _validate_metadata(
        metadata_json: str,
    ) -> bool:

        if not metadata_json:
            print("[OUT] Metadata vacía")
            return False

        try:
            metadata = json.loads(metadata_json)

        except json.JSONDecodeError as exc:
            print(
                f"[OUT] Metadata JSON inválida: {exc}"
            )
            return False

        if not isinstance(metadata, dict):
            print("[OUT] Metadata debe ser un objeto JSON")
            return False

        return True

    # --------------------------------------------------------
    # PROCESAMIENTO
    # --------------------------------------------------------

    async def _call_worker(
        self,
        session: CallSession,
    ):
        """
        Ejecuta processListen() de forma completamente asíncrona.

        IMPORTANTE:
        processListen() debe ser async.
        """

        try:

            print(
                f"[OUT] Iniciando processListen | "
                f"call_id={session.call_id}"
            )

            session.processing = True

            # ------------------------------------------------
            # AQUÍ ESTÁ TU PROCESO REAL
            # ------------------------------------------------

            result = session.processor.processListen(
                session.call_id,
                session.metadata_json,
            )

            # Soporta tanto:
            #
            # async def processListen(...)
            #
            # como una función que devuelva una coroutine.
            #
            if inspect.isawaitable(result):
                await result

            print(
                f"[OUT] processListen terminado | "
                f"call_id={session.call_id}"
            )

        except asyncio.CancelledError:

            print(
                f"[OUT] processListen cancelado | "
                f"call_id={session.call_id}"
            )

            raise

        except Exception as exc:

            print(
                f"[OUT] Error processListen | "
                f"call_id={session.call_id} | "
                f"{type(exc).__name__}: {exc}"
            )

        finally:

            session.processing = False

            # Si durante el procesamiento la sesión quedó
            # marcada para limpieza, se limpia ahora.
            if session.cleanup_requested:
                await self._remove_session_if_inactive(
                    session
                )

    # --------------------------------------------------------
    # VALIDAR ESTADO DEL PROCESSOR
    # --------------------------------------------------------

    @staticmethod
    def _processor_is_active(
        session: CallSession,
    ) -> bool:

        # Nuestro estado interno
        if session.processing:
            return True

        processor = session.processor

        if processor is None:
            return False

        # Si workerProcess tiene status,
        # lo usamos como validación adicional.
        try:

            status = processor.status

            if isinstance(status, bool):
                return status

            if isinstance(status, str):

                status = status.lower().strip()

                return status in {
                    "processing",
                    "active",
                    "running",
                    "busy",
                    "working",
                }

        except Exception:
            pass

        return False

    # --------------------------------------------------------
    # SEND SEGMENT
    # --------------------------------------------------------

    async def SendSegment(
        self,
        request,
        context,
    ):
        """
        Recibe un segmento desde TEN VAD.

        Esta función NO espera a que processListen()
        termine.

        Solo inicia el proceso y responde inmediatamente.
        """

        call_id = request.call_id

        if not call_id:

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        if not request.audio:

            print(
                f"[OUT] Audio vacío | "
                f"call_id={call_id}"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        metadata_json = request.metadata_json

        if not self._validate_metadata(
            metadata_json
        ):

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        # ----------------------------------------------------
        # Obtener sesión
        # ----------------------------------------------------

        session = await self._get_or_create_session(
            call_id
        )

        # ----------------------------------------------------
        # Actualizar actividad
        # ----------------------------------------------------

        async with session.lock:

            session.touch()

            session.cleanup_requested = False

            # Guardamos siempre la metadata más reciente.
            session.metadata_json = metadata_json

            # ------------------------------------------------
            # BLOQUEO OPCIONAL
            # ------------------------------------------------

            if self.block_active:

                if self._processor_is_active(session):

                    print(
                        f"[OUT] IGNORADO | "
                        f"call_id={call_id} | "
                        f"processListen activo"
                    )

                    # El transporte gRPC funcionó correctamente.
                    #
                    # No devolvemos false porque eso podría hacer
                    # que TEN VAD interprete que el OUT falló.
                    return audio_listener_pb2.SegmentResponse(
                        success=True
                    )

                # --------------------------------------------
                # Iniciar proceso
                # --------------------------------------------

                asyncio.create_task(
                    self._call_worker(session)
                )

            else:

                # ------------------------------------------------
                # MODO NO BLOQUEANTE
                # ------------------------------------------------
                #
                # Si el proceso anterior sigue activo,
                # NO usamos el mismo processor.
                #
                # Creamos otro workerProcess para que dos
                # processListen() no compartan estado.
                # ------------------------------------------------

                if self._processor_is_active(session):

                    print(
                        f"[OUT] Nueva ejecución paralela | "
                        f"call_id={call_id}"
                    )

                    processor = workerProcess()

                    asyncio.create_task(
                        self._call_worker_with_processor(
                            session,
                            processor,
                        )
                    )

                else:

                    asyncio.create_task(
                        self._call_worker(session)
                    )

        # ----------------------------------------------------
        # RESPONDER INMEDIATAMENTE
        # ----------------------------------------------------

        return audio_listener_pb2.SegmentResponse(
            success=True
        )

    # --------------------------------------------------------
    # WORKER CON PROCESSOR INDEPENDIENTE
    # --------------------------------------------------------

    async def _call_worker_with_processor(
        self,
        session: CallSession,
        processor,
    ):

        try:

            print(
                f"[OUT] processListen paralelo | "
                f"call_id={session.call_id}"
            )

            result = processor.processListen(
                session.call_id,
                session.metadata_json,
            )

            if inspect.isawaitable(result):
                await result

            print(
                f"[OUT] processListen paralelo terminado | "
                f"call_id={session.call_id}"
            )

        except asyncio.CancelledError:

            raise

        except Exception as exc:

            print(
                f"[OUT] Error processListen paralelo | "
                f"call_id={session.call_id} | "
                f"{type(exc).__name__}: {exc}"
            )

        finally:

            await self._cleanup_processor(
                processor
            )

    # --------------------------------------------------------
    # CLEANUP PROCESSOR
    # --------------------------------------------------------

    @staticmethod
    async def _cleanup_processor(
        processor,
    ):

        if processor is None:
            return

        try:

            # Si tu workerProcess tiene cleanup()
            cleanup = getattr(
                processor,
                "cleanup",
                None,
            )

            if cleanup:

                result = cleanup()

                if inspect.isawaitable(result):
                    await result

                return

            # Alternativa: close()
            close = getattr(
                processor,
                "close",
                None,
            )

            if close:

                result = close()

                if inspect.isawaitable(result):
                    await result

                return

        except Exception as exc:

            print(
                f"[OUT] Error limpiando processor: "
                f"{type(exc).__name__}: {exc}"
            )

    # --------------------------------------------------------
    # REMOVE SESSION
    # --------------------------------------------------------

    async def _remove_session_if_inactive(
        self,
        session: CallSession,
    ):

        async with session.lock:

            # Volvió a tener actividad
            if session.idle_seconds() < SESSION_TIMEOUT_SECONDS:
                session.cleanup_requested = False
                return

            # Todavía procesando
            if self._processor_is_active(session):

                session.cleanup_requested = True

                return

            session_id = session.call_id

            async with self.sessions_lock:

                current = self.sessions.get(
                    session_id
                )

                # Verificamos que siga siendo
                # exactamente la misma sesión.
                if current is not session:
                    return

                self.sessions.pop(
                    session_id,
                    None
                )

            print(
                f"[OUT] Sesión eliminada | "
                f"call_id={session_id} | "
                f"inactiva={session.idle_seconds():.1f}s"
            )

            await self._cleanup_processor(
                session.processor
            )

            session.processor = None

    # --------------------------------------------------------
    # CLEANUP LOOP
    # --------------------------------------------------------

    async def _cleanup_loop(self):

        try:

            while self.running:

                await asyncio.sleep(
                    CLEANUP_INTERVAL_SECONDS
                )

                async with self.sessions_lock:
                    sessions = list(
                        self.sessions.values()
                    )

                # Todas las sesiones se revisan
                # concurrentemente.
                await asyncio.gather(
                    *(
                        self._check_session_timeout(session)
                        for session in sessions
                    ),
                    return_exceptions=True,
                )

        except asyncio.CancelledError:

            raise

    # --------------------------------------------------------
    # CHECK SESSION
    # --------------------------------------------------------

    async def _check_session_timeout(
        self,
        session: CallSession,
    ):

        if session.idle_seconds() < SESSION_TIMEOUT_SECONDS:
            return

        print(
            f"[OUT] Sesión inactiva > "
            f"{SESSION_TIMEOUT_SECONDS}s | "
            f"call_id={session.call_id}"
        )

        await self._remove_session_if_inactive(
            session
        )

    # --------------------------------------------------------
    # PING
    # --------------------------------------------------------

    async def Ping(
        self,
        request,
        context,
    ):

        print(
            f"[OUT] Ping received | "
            f"source={request.source}"
        )

        return audio_listener_pb2.PingResponse(
            success=True,
            service="audio-out",
            message="OUT is alive",
        )


# ============================================================
# SERVIDOR
# ============================================================

async def serve(
    host: str,
    port: int,
    block_active: bool,
):

    service = AudioOutService(
        block_active=block_active
    )

    server = grpc.aio.server()

    audio_listener_pb2_grpc.add_SegmentOutputServicer_to_server(
        service,
        server,
    )

    address = f"{host}:{port}"

    server.add_insecure_port(address)

    await service.start()

    await server.start()

    print(
        "=============================================="
    )

    print(
        "[OUT] Audio OUT iniciado"
    )

    print(
        f"[OUT] Address: {address}"
    )

    print(
        f"[OUT] block_active: {block_active}"
    )

    print(
        f"[OUT] Session timeout: "
        f"{SESSION_TIMEOUT_SECONDS}s"
    )

    print(
        "=============================================="
    )

    # --------------------------------------------------------
    # Shutdown limpio
    # --------------------------------------------------------

    stop_event = asyncio.Event()

    loop = asyncio.get_running_loop()

    def request_shutdown():

        print(
            "\n[OUT] Señal de apagado recibida"
        )

        stop_event.set()

    for sig in (
        signal.SIGINT,
        signal.SIGTERM,
    ):

        try:

            loop.add_signal_handler(
                sig,
                request_shutdown
            )

        except NotImplementedError:
            pass

    try:

        await stop_event.wait()

    finally:

        print(
            "[OUT] Deteniendo gRPC..."
        )

        await server.stop(
            grace=5
        )

        await service.stop()


# ============================================================
# ARGUMENTOS
# ============================================================

def parse_args():

    parser = argparse.ArgumentParser(
        description="Audio OUT asynchronous gRPC service"
    )

    parser.add_argument(
        "--host",
        default="0.0.0.0",
        help="Host donde escucha gRPC",
    )

    parser.add_argument(
        "--port",
        type=int,
        default=9000,
        help="Puerto gRPC",
    )

    parser.add_argument(
        "--block-active",
        action="store_true",
        help=(
            "Ignora nuevas solicitudes de un call_id "
            "mientras processListen está ejecutándose"
        ),
    )

    return parser.parse_args()


# ============================================================
# MAIN
# ============================================================

def main():

    args = parse_args()

    try:

        asyncio.run(
            serve(
                host=args.host,
                port=args.port,
                block_active=args.block_active,
            )
        )

    except KeyboardInterrupt:

        pass


if __name__ == "__main__":
    main()
