def decimal(value: Any) -> Decimal:
    """
    Convierte un valor a Decimal evitando float.
    """
    try:
        return Decimal(str(value))
    except (InvalidOperation, ValueError, TypeError):
        raise ValueError(f"Valor numérico inválido: {value}")


def decimal_to_string(value: Decimal) -> str:
    """
    Representación estable de Decimal.
    """
    value = value.normalize()

    if value == value.to_integral():
        return str(value.quantize(Decimal("1")))

    return format(value, "f")

@tool
def add_numbers(a: str, b: str) -> str:
    """
    Suma dos valores numéricos.

    Usar cuando sea necesario sumar cantidades, dinero,
    saldos, valores, unidades u otros números.
    """
    result = decimal(a) + decimal(b)
    return decimal_to_string(result)


@tool
def subtract_numbers(a: str, b: str) -> str:
    """
    Resta b a a.

    Usar para obtener diferencias entre valores,
    saldos restantes, descuentos, pagos, etc.
    """
    result = decimal(a) - decimal(b)
    return decimal_to_string(result)


@tool
def multiply_numbers(a: str, b: str) -> str:
    """
    Multiplica dos valores numéricos.
    """
    result = decimal(a) * decimal(b)
    return decimal_to_string(result)


@tool
def divide_numbers(a: str, b: str) -> str:
    """
    Divide a entre b.
    """
    divisor = decimal(b)

    if divisor == 0:
        raise ValueError("No se puede dividir entre cero.")

    result = decimal(a) / divisor

    return decimal_to_string(result)


@tool
def absolute_difference(a: str, b: str) -> str:
    """
    Obtiene la diferencia absoluta entre dos números.
    """
    result = abs(decimal(a) - decimal(b))
    return decimal_to_string(result)


@tool
def remaining_balance(
    original_balance: str,
    payment: str,
) -> str:
    """
    Calcula el saldo restante después de un pago.

    Fórmula:
        saldo restante = saldo original - pago
    """
    balance = decimal(original_balance)
    paid = decimal(payment)

    result = balance - paid

    return decimal_to_string(result)


@tool
def percentage_of(
    percentage: str,
    value: str,
) -> str:
    """
    Calcula un porcentaje de un valor.

    Ejemplo:
        10% de 500 = 50
    """
    result = (decimal(percentage) / Decimal("100")) * decimal(value)

    return decimal_to_string(result)


@tool
def calculate_percentage(
    part: str,
    total: str,
) -> str:
    """
    Calcula qué porcentaje representa part sobre total.

    Ejemplo:
        50 de 200 = 25%
    """
    total_value = decimal(total)

    if total_value == 0:
        raise ValueError("El total no puede ser cero.")

    result = (decimal(part) / total_value) * Decimal("100")

    return decimal_to_string(result)


@tool
def round_number(
    value: str,
    decimals: int = 2,
) -> str:
    """
    Redondea un número a la cantidad de decimales indicada.
    """
    if decimals < 0 or decimals > 10:
        raise ValueError("decimals debe estar entre 0 y 10.")

    quantizer = Decimal("1").scaleb(-decimals)

    result = decimal(value).quantize(
        quantizer,
        rounding=ROUND_HALF_UP,
    )

    return decimal_to_string(result)

@tool
def compare_numbers(
    a: str,
    b: str,
) -> str:
    """
    Compara dos números.

    Devuelve:
        greater
        less
        equal
    """
    value_a = decimal(a)
    value_b = decimal(b)

    if value_a > value_b:
        return "greater"

    if value_a < value_b:
        return "less"

    return "equal"

def parse_date(value: str) -> date:
    try:
        return date.fromisoformat(value)
    except ValueError:
        raise ValueError(
            f"Fecha inválida: {value}. "
            "Utiliza formato YYYY-MM-DD."
        )


@tool
def add_days(
    input_date: str,
    days: int,
) -> str:
    """
    Agrega una cantidad de días a una fecha.

    Formato:
        YYYY-MM-DD
    """
    parsed = parse_date(input_date)

    result = parsed + timedelta(days=days)

    return result.isoformat()


@tool
def subtract_days(
    input_date: str,
    days: int,
) -> str:
    """
    Resta una cantidad de días a una fecha.
    """
    parsed = parse_date(input_date)

    result = parsed - timedelta(days=days)

    return result.isoformat()


@tool
def days_between_dates(
    start_date: str,
    end_date: str,
) -> str:
    """
    Calcula la diferencia de días entre dos fechas.

    Resultado:
        end_date - start_date

    Puede ser positivo, cero o negativo.
    """
    start = parse_date(start_date)
    end = parse_date(end_date)

    return str((end - start).days)


@tool
def days_until_date(
    current_date: str,
    target_date: str,
) -> str:
    """
    Calcula cuántos días faltan desde current_date
    hasta target_date.

    Positivo:
        faltan días.

    Cero:
        es hoy.

    Negativo:
        la fecha ya pasó.
    """
    current = parse_date(current_date)
    target = parse_date(target_date)

    return str((target - current).days)


@tool
def days_overdue(
    current_date: str,
    due_date: str,
) -> str:
    """
    Calcula cuántos días lleva vencida una fecha.

    Si todavía no está vencida devuelve 0.
    """
    current = parse_date(current_date)
    due = parse_date(due_date)

    difference = (current - due).days

    return str(max(0, difference))


@tool
def add_months(
    input_date: str,
    months: int,
) -> str:
    """
    Agrega meses a una fecha.

    Ajusta automáticamente el día cuando el mes destino
    tiene menos días.
    """
    original = parse_date(input_date)

    month_index = original.month - 1 + months

    year = original.year + month_index // 12
    month = month_index % 12 + 1

    # Último día del mes destino
    if month == 12:
        next_month = date(year + 1, 1, 1)
    else:
        next_month = date(year, month + 1, 1)

    last_day = (next_month - timedelta(days=1)).day

    day = min(original.day, last_day)

    return date(year, month, day).isoformat()


@tool
def subtract_months(
    input_date: str,
    months: int,
) -> str:
    """
    Resta meses a una fecha.
    """
    return add_months.invoke(
        {
            "input_date": input_date,
            "months": -months,
        }
    )


@tool
def add_years(
    input_date: str,
    years: int,
) -> str:
    """
    Agrega años a una fecha.
    """
    original = parse_date(input_date)

    target_year = original.year + years

    try:
        result = original.replace(year=target_year)
    except ValueError:
        # 29 de febrero
        result = original.replace(
            year=target_year,
            day=28,
        )

    return result.isoformat()


@tool
def subtract_years(
    input_date: str,
    years: int,
) -> str:
    """
    Resta años a una fecha.
    """
    return add_years.invoke(
        {
            "input_date": input_date,
            "years": -years,
        }
    )

class AgentState(TypedDict):
    messages: list[BaseMessage]
    tool_iterations: int


# ============================================================
# AGENTE
# ============================================================

class BusinessAgent:

    def __init__(
        self,
        *,
        model: str,
        base_url: str,
        api_key: str,
        temperature: float,
        max_tokens: int,
        timeout: float,
        max_tool_iterations: int,
    ):
        self.model = model
        self.max_tool_iterations = max_tool_iterations

        logger.info("Inicializando LLM")
        logger.info("model=%s", model)
        logger.info("base_url=%s", base_url)
        logger.info("temperature=%s", temperature)
        logger.info("max_tokens=%s", max_tokens)

        self.llm = ChatOpenAI(
            model=model,
            base_url=base_url,
            api_key=api_key,
            temperature=temperature,
            max_tokens=max_tokens,
            timeout=timeout,
            max_retries=2,
        )

        self.llm_with_tools = self.llm.bind_tools(TOOLS)

        self.tool_node = ToolNode(TOOLS)

        self.graph = self._build_graph()

    # --------------------------------------------------------
    # GRAPH
    # --------------------------------------------------------

    def _build_graph(self):

        graph = StateGraph(AgentState)

        graph.add_node(
            "agent",
            self.agent_node,
        )

        graph.add_node(
            "tools",
            self.tool_node,
        )

        graph.add_edge(
            START,
            "agent",
        )

        graph.add_conditional_edges(
            "agent",
            self.should_continue,
            {
                "tools": "tools",
                "end": END,
            },
        )

        graph.add_edge(
            "tools",
            "agent",
        )

        return graph.compile()

    # --------------------------------------------------------
    # AGENT NODE
    # --------------------------------------------------------

    def agent_node(
        self,
        state: AgentState,
    ) -> dict:

        messages = state["messages"]

        tool_iterations = state.get(
            "tool_iterations",
            0,
        )

        if tool_iterations >= self.max_tool_iterations:
            logger.warning(
                "Maximum tool iterations reached"
            )

            # Forzamos una última respuesta sin herramientas.
            response = self.llm.invoke(messages)

            return {
                "messages": [response],
                "tool_iterations": tool_iterations,
            }

        response = self.llm_with_tools.invoke(messages)

        next_iterations = tool_iterations

        if isinstance(response, AIMessage):
            if response.tool_calls:
                next_iterations += 1

        return {
            "messages": [response],
            "tool_iterations": next_iterations,
        }

    # --------------------------------------------------------
    # ROUTER
    # --------------------------------------------------------

    def should_continue(
        self,
        state: AgentState,
    ) -> Literal["tools", "end"]:

        messages = state["messages"]

        if not messages:
            return "end"

        last_message = messages[-1]

        if isinstance(last_message, AIMessage):
            if last_message.tool_calls:
                if (
                    state.get("tool_iterations", 0)
                    < self.max_tool_iterations
                ):
                    return "tools"

        return "end"

    # --------------------------------------------------------
    # INVOCACIÓN
    # --------------------------------------------------------

    async def ainvoke(
        self,
        messages: list[BaseMessage],
    ) -> str:

        initial_state: AgentState = {
            "messages": messages,
            "tool_iterations": 0,
        }

        result = await self.graph.ainvoke(
            initial_state,
            config={
                "recursion_limit": (
                    self.max_tool_iterations * 3 + 5
                )
            },
        )

        final_messages = result["messages"]

        for message in reversed(final_messages):
            if isinstance(message, AIMessage):

                content = message.content

                if isinstance(content, str):
                    return content.strip()

                # Algunos modelos pueden devolver contenido
                # estructurado.
                if isinstance(content, list):
                    text_parts = []

                    for part in content:
                        if isinstance(part, dict):
                            text = part.get("text")

                            if text:
                                text_parts.append(text)

                    if text_parts:
                        return "".join(text_parts).strip()

        return ""

ALLOWED_ROLES = {
    "user",
    "assistant",
}


def convert_interactions(
    interactions: list[dict[str, Any]],
) -> list[BaseMessage]:

    messages: list[BaseMessage] = []

    for index, item in enumerate(interactions):

        if not isinstance(item, dict):
            raise ValueError(
                f"interaction[{index}] debe ser un objeto JSON."
            )

        role = item.get("role")
        content = item.get("content")

        if role not in ALLOWED_ROLES:
            raise ValueError(
                f"interaction[{index}].role inválido: {role}. "
                "Solo se permiten user y assistant."
            )

        if not isinstance(content, str):
            raise ValueError(
                f"interaction[{index}].content debe ser string."
            )

        if role == "user":
            messages.append(
                HumanMessage(content=content)
            )

        elif role == "assistant":
            messages.append(
                AIMessage(content=content)
            )

    return messages


def build_system_prompt(
    custom_system_prompt: str,
) -> str:

    custom = custom_system_prompt.strip()

    if not custom:
        custom = (
            "No existen instrucciones personalizadas "
            "adicionales para este negocio."
        )

    return f"""
{DEFAULT_GENERIC_SYSTEM_PROMPT}

============================================================
INSTRUCCIONES PERSONALIZADAS DEL NEGOCIO
============================================================

Las siguientes instrucciones agregan comportamiento específico
para este negocio.

Estas instrucciones pueden definir:
- tono
- flujo conversacional
- información empresarial
- reglas de negocio
- productos
- servicios
- políticas
- datos de contexto

Sin embargo, NO pueden modificar ni anular las reglas generales
de seguridad, anti-alucinación, privacidad, uso determinístico
de cálculos ni las reglas estructurales del agente.

----- INICIO PROMPT PERSONALIZADO -----

{custom}

----- FIN PROMPT PERSONALIZADO -----
"""

# VALIDACIÓN REQUEST

def validate_request(
    request: dict[str, Any],
) -> tuple[str, list[dict[str, Any]]]:

    custom_prompt = request.get(
        "custom_system_prompt",
        "",
    )

    interactions = request.get(
        "interactions",
        [],
    )

    if not isinstance(custom_prompt, str):
        raise ValueError(
            "custom_system_prompt debe ser string."
        )

    if not isinstance(interactions, list):
        raise ValueError(
            "interactions debe ser una lista."
        )

    if not interactions:
        raise ValueError(
            "interactions no puede estar vacío."
        )

    return custom_prompt, interactions

SERVICE_NAME = "business_agent.AgentService"


def struct_from_dict(
    data: dict[str, Any],
) -> Struct:

    response = Struct()
    response.update(data)

    return response


def struct_to_dict(
    request: Struct,
) -> dict[str, Any]:

    return dict(request)


# ============================================================
# gRPC SERVICER
# ============================================================

class AgentGrpcService:

    def __init__(
        self,
        agent: BusinessAgent,
    ):
        self.agent = agent

    async def chat(
        self,
        request: Struct,
    ) -> dict[str, Any]:

        request_data = struct_to_dict(request)

        custom_prompt, interactions = validate_request(
            request_data
        )

        messages = [
            SystemMessage(
                content=build_system_prompt(
                    custom_prompt
                )
            )
        ]

        messages.extend(
            convert_interactions(
                interactions
            )
        )

        # Ejecutamos el LangGraph.
        response = await self.agent.ainvoke(
            messages
        )

        if not response:
            raise RuntimeError(
                "El agente no produjo una respuesta."
            )

        return {
            "ok": True,
            "response": response,
        }


# ============================================================
# gRPC METHOD HANDLERS
# ============================================================

async def grpc_chat(
    service: AgentGrpcService,
    request: Struct,
    context: grpc.aio.ServicerContext,
) -> Struct:

    try:

        result = await service.chat(request)

        return struct_from_dict(result)

    except Exception as exc:

        logger.exception(
            "Error procesando request"
        )

        await context.abort(
            grpc.StatusCode.INVALID_ARGUMENT,
            str(exc),
        )


async def grpc_health(
    service: AgentGrpcService,
    request: Struct,
    context: grpc.aio.ServicerContext,
) -> Struct:

    return struct_from_dict(
        {
            "ok": True,
            "service": "business-agent",
            "stateless": True,
        }
    )


# ============================================================
# SERVER
# ============================================================

async def serve(
    host: str,
    port: int,
    agent: BusinessAgent,
    grpc_workers: int,
):

    server = grpc.aio.server(
        options=[
            (
                "grpc.max_receive_message_length",
                16 * 1024 * 1024,
            ),
            (
                "grpc.max_send_message_length",
                16 * 1024 * 1024,
            ),
        ]
    )

    service = AgentGrpcService(agent)

    handlers = {
        "Chat": grpc.unary_unary_rpc_method_handler(
            lambda request, context:
                grpc_chat(
                    service,
                    request,
                    context,
                ),
            request_deserializer=Struct.FromString,
            response_serializer=Struct.SerializeToString,
        ),

        "Health": grpc.unary_unary_rpc_method_handler(
            lambda request, context:
                grpc_health(
                    service,
                    request,
                    context,
                ),
            request_deserializer=Struct.FromString,
            response_serializer=Struct.SerializeToString,
        ),
    }

    generic_handler = grpc.method_handlers_generic_handler(
        SERVICE_NAME,
        handlers,
    )

    server.add_generic_rpc_handlers(
        (generic_handler,)
    )

    address = f"{host}:{port}"

    server.add_insecure_port(address)

    logger.info(
        "=================================================="
    )
    logger.info(
        "Business Agent gRPC iniciado"
    )
    logger.info(
        "Address: %s",
        address,
    )
    logger.info(
        "Stateless: TRUE"
    )
    logger.info(
        "Workers: %s",
        grpc_workers,
    )
    logger.info(
        "=================================================="
    )

    await server.start()

    await server.wait_for_termination()

                ),

### Lo importante del request

Cada llamada lleva absolutamente todo:

```json
{
  "custom_system_prompt": "Eres un agente de cobranza. Sé breve y profesional.",
  "interactions": [
    {
      "role": "user",
      "content": "Mi deuda actual es de 300."
    },
    {
      "role": "assistant",
      "content": "Entiendo."
    },
    {
      "role": "user",
      "content": "Ya pagué 200, ¿cuánto me queda?"
    }
  ]
}
```

El servidor construye internamente:

```text
SYSTEM
 ├── reglas genéricas
 └── prompt personalizado

USER
 └── historial completo recibido

        ↓

     LangGraph
        ↓
     LLM decide
        ↓
   ¿necesita cálculo?
      /       \
    NO         SÍ
    ↓          ↓
 respuesta   Python Tool
                 ↓
             resultado
                 ↓
               LLM
                 ↓
             respuesta
```

Y **después de terminar la solicitud no queda nada almacenado**.

No existe:

```text
session_id
memory
Redis
DB
conversation store
hash
cache de conversaciones
```

El estado que utiliza LangGraph durante una llamada es solamente el estado temporal necesario para ejecutar **esa solicitud**.

### Un detalle importante para producción

Yo mantendría `temperature=0` para este servicio. Incluso si posteriormente quieres respuestas un poco más naturales, puedes probar:

```bash
--temperature 0.1
```

o:

```bash
--temperature 0.2
```

pero no subiría mucho la temperatura en un agente donde necesitamos que **detecte correctamente cuándo debe usar una herramienta y respete estrictamente el contexto**.

También dejaría el límite:

```bash
--max-tool-iterations 5
```

para evitar que un modelo entre en un ciclo del tipo:

```text
LLM → tool → LLM → tool → LLM → tool → ...
```

sin afectar a las solicitudes normales.

**Una mejora que haría como siguiente paso** es agregar al mismo script un modo de **respuesta estructurada/JSON opcional**, además de `response` de texto, para que tu aplicación de voz pueda recibir algo como `response`, `intent`, `requires_data`, `calculation_used`, etc., sin que eso implique guardar estado.

-----------------------------------------------------------

Sí. Para lo que estás construyendo, **es mejor tener un `.proto` real** y generar los contratos Python de cliente y servidor. Así quedan claramente definidos los tipos de request/response y puedes consumir el servicio desde Python, C++, Java, etc.

Te propongo esta estructura:

```text
agent/
├── agent_server.py
├── agent_client.py
├── agent.proto
├── agent_pb2.py
└── agent_pb2_grpc.py
```

El `.proto` será el contrato y `agent_server.py` / `agent_client.py` serán las implementaciones.

### 1. `agent.proto`

```proto
syntax = "proto3";

package business_agent;


// ============================================================
// SERVICIO
// ============================================================

service BusinessAgentService {

  rpc Chat(ChatRequest) returns (ChatResponse);

  rpc Health(HealthRequest) returns (HealthResponse);
}


// ============================================================
// INTERACCIONES
// ============================================================

message Interaction {
  string role = 1;
  string content = 2;
}


// ============================================================
// CHAT REQUEST
// ============================================================

message ChatRequest {

  // Prompt específico del negocio.
  string custom_system_prompt = 1;

  // Historial completo de la conversación.
  repeated Interaction interactions = 2;
}


// ============================================================
// CHAT RESPONSE
// ============================================================

message ChatResponse {

  bool ok = 1;

  // Respuesta final del agente.
  string response = 2;

  // Mensaje de error si ok = false.
  string error = 3;
}


// ============================================================
// HEALTH
// ============================================================

message HealthRequest {
}


message HealthResponse {

  bool ok = 1;

  string service = 2;

  bool stateless = 3;
}
```

La parte importante es que **no existe `session_id`**.

Cada request es completamente independiente.

# 3. `agent_client.py`

Este sería el cliente real:

```python
#!/usr/bin/env python3

import argparse
import grpc

import agent_pb2
import agent_pb2_grpc


def build_request(
    custom_system_prompt: str,
    interactions: list[dict[str, str]],
) -> agent_pb2.ChatRequest:

    request = agent_pb2.ChatRequest()

    request.custom_system_prompt = custom_system_prompt

    for interaction in interactions:

        item = request.interactions.add()

        item.role = interaction["role"]
        item.content = interaction["content"]

    return request


def chat(
    address: str,
    custom_system_prompt: str,
    interactions: list[dict[str, str]],
    timeout: float,
):

    with grpc.insecure_channel(address) as channel:

        stub = agent_pb2_grpc.BusinessAgentServiceStub(
            channel
        )

        request = build_request(
            custom_system_prompt=custom_system_prompt,
            interactions=interactions,
        )

        response = stub.Chat(
            request,
            timeout=timeout,
        )

        if not response.ok:

            raise RuntimeError(
                response.error
            )

        return response.response


def health(
    address: str,
    timeout: float,
):

    with grpc.insecure_channel(address) as channel:

        stub = agent_pb2_grpc.BusinessAgentServiceStub(
            channel
        )

        response = stub.Health(
            agent_pb2.HealthRequest(),
            timeout=timeout,
        )

        print(
            f"ok={response.ok}"
        )

        print(
            f"service={response.service}"
        )

        print(
            f"stateless={response.stateless}"
        )


def main():

    parser = argparse.ArgumentParser()

    parser.add_argument(
        "--address",
        default="127.0.0.1:50051",
    )

    parser.add_argument(
        "--timeout",
        type=float,
        default=120,
    )

    parser.add_argument(
        "--health",
        action="store_true",
    )

    args = parser.parse_args()

    if args.health:

        health(
            address=args.address,
            timeout=args.timeout,
        )

        return

    custom_prompt = """
Eres un agente de cobranza.

Responde de manera profesional,
clara y breve.

Si el usuario pregunta por su deuda,
utiliza los datos disponibles en el contexto.
"""

    interactions = [

        {
            "role": "user",
            "content": (
                "Mi deuda actual es de 300."
            ),
        },

        {
            "role": "assistant",
            "content": (
                "Entiendo, tienes una deuda "
                "registrada de 300."
            ),
        },

        {
            "role": "user",
            "content": (
                "Ya pagué 200, ¿cuánto me queda?"
            ),
        },

    ]

    response = chat(
        address=args.address,
        custom_system_prompt=custom_prompt,
        interactions=interactions,
        timeout=args.timeout,
    )

    print()
    print("RESPUESTA:")
    print(response)


if __name__ == "__main__":
    main()
```

# 4. `agent_server.py`

Aquí sí cambiaríamos el servidor que te pasé anteriormente: **eliminamos todo el mecanismo genérico de `Struct`** y usamos directamente los mensajes generados por protobuf.

Los imports quedarían así:

```python
import grpc

import agent_pb2
import agent_pb2_grpc
```

Y el servicio:

```python
class BusinessAgentGrpcService(
    agent_pb2_grpc.BusinessAgentServiceServicer
):

    def __init__(self, agent):
        self.agent = agent

    async def Chat(
        self,
        request,
        context,
    ):

        try:

            if not request.interactions:

                await context.abort(
                    grpc.StatusCode.INVALID_ARGUMENT,
                    "interactions no puede estar vacío.",
                )

            messages = [
                SystemMessage(
                    content=build_system_prompt(
                        request.custom_system_prompt
                    )
                )
            ]

            for interaction in request.interactions:

                if interaction.role == "user":

                    messages.append(
                        HumanMessage(
                            content=interaction.content
                        )
                    )

                elif interaction.role == "assistant":

                    messages.append(
                        AIMessage(
                            content=interaction.content
                        )
                    )

                else:

                    await context.abort(
                        grpc.StatusCode.INVALID_ARGUMENT,
                        (
                            f"Role inválido: "
                            f"{interaction.role}"
                        ),
                    )

            response = await self.agent.ainvoke(
                messages
            )

            return agent_pb2.ChatResponse(
                ok=True,
                response=response,
                error="",
            )

        except grpc.aio.AbortError:
            raise

        except Exception as exc:

            logger.exception(
                "Error procesando Chat"
            )

            return agent_pb2.ChatResponse(
                ok=False,
                response="",
                error=str(exc),
            )

    async def Health(
        self,
        request,
        context,
    ):

        return agent_pb2.HealthResponse(
            ok=True,
            service="business-agent",
            stateless=True,
        )
```

Y el servidor:

```python
async def serve(
    host: str,
    port: int,
    agent: BusinessAgent,
):

    server = grpc.aio.server()

    agent_pb2_grpc.add_BusinessAgentServiceServicer_to_server(
        BusinessAgentGrpcService(agent),
        server,
    )

    address = f"{host}:{port}"

    server.add_insecure_port(address)

    logger.info(
        "Business Agent escuchando en %s",
        address,
    )

    await server.start()

    await server.wait_for_termination()
```

## Flujo final

Quedaría bastante limpio:

```text
                 agent.proto
                     │
          ┌──────────┴──────────┐
          ↓                     ↓
   agent_pb2.py          agent_pb2_grpc.py
          │                     │
          │              ┌──────┴──────┐
          │              ↓             ↓
          │        agent_server.py  agent_client.py
          │              │             │
          │              ↓             │
          │          LangGraph         │
          │              │             │
          │              ↓             │
          │         LLM/vLLM/Ollama    │
          │              │             │
          └──────────────┬──────────────┘
                         │
                       gRPC
```

Y conceptualmente:

```text
CLIENT
  │
  │ ChatRequest
  │
  ├── custom_system_prompt
  │
  └── interactions[]
          │
          ↓
       gRPC
          │
          ↓
SERVER
  │
  ├── generic prompt
  ├── custom prompt
  ├── interaction history
  │
  ↓
LangGraph
  │
  ├── LLM
  │
  └── Python Tools
       ├── fechas
       ├── días
       ├── sumas
       ├── restas
       ├── saldos
       ├── porcentajes
       └── etc.
  │
  ↓
ChatResponse
  │
  └── response
```

### Y esto tiene una ventaja importante

El contrato queda independiente del modelo.

El cliente **no necesita saber** si detrás tienes:

```text
vLLM
Ollama
Qwen
gpt-oss
otro modelo
otra GPU
otro servidor
```

Solo conoce:

```proto
rpc Chat(ChatRequest) returns (ChatResponse);
```

Además, si posteriormente quieres consumir este agente desde tu **servicio C++ del voicebot**, ese mismo `agent.proto` puede generar los stubs C++:

```bash
protoc \
    -I. \
    --cpp_out=. \
    --grpc_out=. \
    --plugin=protoc-gen-grpc=$(which grpc_cpp_plugin) \
    agent.proto
```

y tendrías:

```text
agent.pb.h
agent.pb.cc
agent.grpc.pb.h
agent.grpc.pb.cc
```

Así tienes **un único contrato `.proto` como fuente de verdad para todos los clientes**, que para tu arquitectura de servicios es bastante más conveniente que el `Struct` genérico.
