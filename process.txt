UDP_HEADER_SIZE = 4

# Tamaño máximo esperado de un datagrama UDP.
# Ajustable según el tamaño de los paquetes que envíe audio_headset.py.
UDP_MAX_PACKET_SIZE = 65535

# Buffer de recepción del socket.
UDP_RECEIVE_BUFFER = 16 * 1024 * 1024

# Timeout para conectar inicialmente con un TEN VAD.
VAD_CONNECT_TIMEOUT = 10.0

# Tiempo máximo sin recibir audio antes de liberar automáticamente
# una llamada.
DEFAULT_CALL_TIMEOUT = 30.0

# Tiempo entre comprobaciones del timeout.
DEFAULT_CLEANUP_INTERVAL = 5.0

# Máximo de elementos pendientes.
#
# Se deja configurable para evitar crecimiento infinito de memoria.
DEFAULT_MAX_PENDING = 10000


# ===========================================================================
# UTILIDADES
# ===========================================================================

def encode_metadata(metadata):
    """
    Convierte metadata a JSON compacto.

    Esta función se mantiene síncrona porque solamente realiza una
    transformación pequeña de datos y no realiza I/O.
    """

    if metadata is None:
        return "{}"

    if isinstance(metadata, str):
        try:
            parsed = json.loads(metadata)

            return json.dumps(
                parsed,
                ensure_ascii=False,
                separators=(",", ":"),
            )

        except json.JSONDecodeError:
            return metadata

    return json.dumps(
        metadata,
        ensure_ascii=False,
        separators=(",", ":"),
    )


def decode_udp_packet(data):
    """
    Decodifica un paquete UDP.

    Formato:

        [4 bytes]
            longitud del header JSON

        [N bytes]
            header JSON UTF-8

        [resto]
            audio PCM

    Header esperado:

        {
            "call_id": "...",
            "metadata": {...}
        }

    También soporta "metadata_json" como string.
    """

    if len(data) < UDP_HEADER_SIZE:
        raise ValueError(
            "Paquete UDP demasiado pequeño para contener el header"
        )

    header_length = struct.unpack(
        "!I",
        data[:UDP_HEADER_SIZE],
    )[0]

    if header_length <= 0:
        raise ValueError(
            "Longitud de header UDP inválida"
        )

    header_start = UDP_HEADER_SIZE
    header_end = header_start + header_length

    if header_end > len(data):
        raise ValueError(
            "Header UDP excede el tamaño del paquete"
        )

    header_bytes = data[
        header_start:header_end
    ]

    audio = data[header_end:]

    try:
        header = json.loads(
            header_bytes.decode("utf-8")
        )
    except (UnicodeDecodeError, json.JSONDecodeError) as exc:
        raise ValueError(
            f"Header JSON inválido: {exc}"
        ) from exc

    call_id = header.get("call_id")

    if not call_id:
        raise ValueError(
            "El paquete UDP no contiene call_id"
        )

    metadata = header.get("metadata_json")

    if metadata is None:
        metadata = header.get("metadata", {})

    metadata_json = encode_metadata(metadata)

    return (
        str(call_id),
        metadata_json,
        audio,
    )

class VadSession:
    """
    Representa una llamada conectada a UN TEN VAD.

    Cada llamada tiene:

        call_id
            Identificador de la llamada.

        vad_server
            Dirección del TEN VAD asignado.

        out_host
            Host del servidor OUT.

        out_port
            Puerto del servidor OUT.

        queue
            Cola independiente de audio.

        grpc_channel
            Canal gRPC independiente.

        task
            Task asyncio encargada de mantener el stream abierto.

    IMPORTANTE:

    Nunca existe una cola global de audio.

    Cada call_id tiene su propia cola:

        call A -> queue A -> VAD 12001 -> OUT 9001

        call B -> queue B -> VAD 12002 -> OUT 9002

        call C -> queue C -> VAD 12003 -> OUT 9003
    """

    def __init__(
        self,
        call_id,
        vad_server,
        out_host,
        out_port,
        call_timeout,
    ):
        self.call_id = call_id

        self.vad_server = vad_server

        self.out_host = out_host
        self.out_port = out_port

        self.call_timeout = call_timeout

        # Cola exclusiva de esta llamada.
        self.queue = asyncio.Queue()

        # Canal gRPC.
        self.channel = None

        # Stub generado.
        self.stub = None

        # Task principal de la sesión.
        self.task = None

        # Indica si la sesión sigue activa.
        self.running = True

        # Último momento en que llegó audio.
        self.last_audio = time.monotonic()

        # Última metadata recibida.
        self.metadata_json = "{}"

        # Evita iniciar dos veces la misma sesión.
        self.started = False

    # -----------------------------------------------------------------------
    # START
    # -----------------------------------------------------------------------

    async def start(self):
        """
        Inicia la task que mantiene la conexión gRPC con TEN VAD.
        """

        if self.started:
            return

        self.started = True

        self.task = asyncio.create_task(
            self._run(),
            name=f"vad-session-{self.call_id}",
        )

    # -----------------------------------------------------------------------
    # PUSH AUDIO
    # -----------------------------------------------------------------------

    async def push_audio(
        self,
        audio,
        metadata_json,
    ):
        """
        Agrega audio a la cola de esta llamada.

        No espera a que TEN VAD procese el audio.

        Esto es importante para que el listener UDP pueda seguir
        recibiendo paquetes.
        """

        if not self.running:
            return

        self.last_audio = time.monotonic()

        # ---------------------------------------------------------------
        # Procesamos metadata
        # ---------------------------------------------------------------

        try:
            metadata = json.loads(
                metadata_json
            )

            if not isinstance(metadata, dict):
                metadata = {}

        except json.JSONDecodeError:
            metadata = {}

        # ---------------------------------------------------------------
        # El manager conoce el OUT que corresponde al VAD.
        #
        # Por seguridad sobrescribimos estos valores para garantizar:
        #
        # VAD 12001 -> OUT 9001
        # VAD 12002 -> OUT 9002
        # ...
        # ---------------------------------------------------------------

        metadata["out_host"] = self.out_host
        metadata["out_port"] = self.out_port

        self.metadata_json = json.dumps(
            metadata,
            ensure_ascii=False,
            separators=(",", ":"),
        )
        
        if self.call_id == "call-001":
            with open("debug_call-001.pcm", "ab") as f:
                f.write(audio)

        # ---------------------------------------------------------------
        # Agregamos el audio a la cola de esta llamada.
        # ---------------------------------------------------------------

        await self.queue.put(
            (
                bytes(audio),
                self.metadata_json,
            )
        )

    # -----------------------------------------------------------------------
    # STOP
    # -----------------------------------------------------------------------

    async def stop(self):
        """
        Detiene la sesión.

        Se coloca None en la cola para despertar al generator
        si está esperando audio.
        """

        if not self.running:
            return

        self.running = False

        # Despierta el generator si está bloqueado esperando datos.
        await self.queue.put(None)

        # Esperamos que termine la task principal.
        if self.task is not None:

            try:
                await asyncio.wait_for(
                    self.task,
                    timeout=5.0,
                )

            except asyncio.TimeoutError:

                self.task.cancel()

                try:
                    await self.task
                except asyncio.CancelledError:
                    pass

            except asyncio.CancelledError:
                pass

        self.task = None

        # Cerramos el canal gRPC.
        if self.channel is not None:

            try:
                await self.channel.close()
            except Exception:
                pass

            self.channel = None

    # -----------------------------------------------------------------------
    # GENERATOR ASÍNCRONO
    # -----------------------------------------------------------------------

    async def _generator(self):
        """
        Generator asíncrono utilizado por grpc.aio.

        Va produciendo AudioStreamRequest conforme llega audio.

        IMPORTANTE:

        Es un generator async:

            async def
            yield

        No es una función normal.
        """

        while self.running:

            item = await self.queue.get()

            # None significa que debemos terminar.
            if item is None:
                break

            audio, metadata_json = item

            request = audio_listener_pb2.AudioStreamRequest(
                call_id=self.call_id,
                audio=audio,
                metadata_json=metadata_json,
            )

            yield request

    # -----------------------------------------------------------------------
    # RUN
    # -----------------------------------------------------------------------

    async def _run(self):
        """
        Mantiene abierto el stream gRPC hacia TEN VAD.

        StreamAudio es CLIENT-STREAMING:

            cliente:
                AudioStreamRequest
                AudioStreamRequest
                AudioStreamRequest
                        ...
                        ↓
                    TEN VAD
                        ↓
                StreamResponse

        Por lo tanto NO usamos:

            async for response in responses

        sino:

            response = await call
        """

        self.channel = grpc.aio.insecure_channel(
            self.vad_server
        )

        try:

            # ---------------------------------------------------------------
            # Esperamos que TEN VAD esté disponible.
            # ---------------------------------------------------------------

            await asyncio.wait_for(
                self.channel.channel_ready(),
                timeout=VAD_CONNECT_TIMEOUT,
            )

            if not self.running:
                return

            self.stub = (
                audio_listener_pb2_grpc.TenVadStub(
                    self.channel
                )
            )

            print(
                f"[VAD] Conectado | "
                f"call_id={self.call_id} | "
                f"vad={self.vad_server} | "
                f"out={self.out_host}:{self.out_port}"
            )

            # ---------------------------------------------------------------
            # STREAM CLIENT-STREAMING
            #
            # _generator() produce:
            #
            #   AudioStreamRequest
            #   AudioStreamRequest
            #   AudioStreamRequest
            #
            # El servidor devuelve UNA sola respuesta.
            # ---------------------------------------------------------------

            call = self.stub.StreamAudio(
                self._generator()
            )

            response = await call

            if not response.success:

                print(
                    f"[VAD] Stream reportó fallo | "
                    f"call_id={self.call_id}"
                )

            else:

                print(
                    f"[VAD] Stream finalizado correctamente | "
                    f"call_id={self.call_id}"
                )

        except asyncio.CancelledError:

            raise

        except grpc.aio.AioRpcError as exc:

            if self.running:

                print(
                    f"[VAD] Error gRPC | "
                    f"call_id={self.call_id} | "
                    f"vad={self.vad_server} | "
                    f"code={exc.code()} | "
                    f"details={exc.details()}"
                )

        except asyncio.TimeoutError:

            if self.running:

                print(
                    f"[VAD] Timeout conectando | "
                    f"call_id={self.call_id} | "
                    f"vad={self.vad_server}"
                )

        except Exception as exc:

            if self.running:

                print(
                    f"[VAD] Error inesperado | "
                    f"call_id={self.call_id} | "
                    f"error={exc}"
                )

        finally:

            if self.channel is not None:

                try:
                    await self.channel.close()
                except Exception:
                    pass

                self.channel = None

            print(
                f"[VAD] Sesión finalizada | "
                f"call_id={self.call_id} | "
                f"vad={self.vad_server}"
            )

# UDP SERVER

class UDPServer:
    """
    Servidor UDP asíncrono.

    Escucha UN solo puerto:

        0.0.0.0:12000

    No utiliza threads.

    Utiliza:

        asyncio loop
        loop.sock_recvfrom()

    De esta manera evitamos DatagramProtocol y podemos mantener
    toda la lógica propia como async def.
    """

    def __init__(
        self,
        host,
        port,
        manager,
    ):
        self.host = host
        self.port = port

        self.manager = manager

        self.socket = None

        self.task = None

        self.running = False

    # -----------------------------------------------------------------------
    # START
    # -----------------------------------------------------------------------

    async def start(self):
        """
        Crea y empieza a escuchar el socket UDP.
        """

        if self.running:
            return

        self.socket = socket.socket(
            socket.AF_INET,
            socket.SOCK_DGRAM,
        )

        # Permite reutilizar el puerto después de reiniciar.
        self.socket.setsockopt(
            socket.SOL_SOCKET,
            socket.SO_REUSEADDR,
            1,
        )

        # Aumentamos el buffer del kernel.
        self.socket.setsockopt(
            socket.SOL_SOCKET,
            socket.SO_RCVBUF,
            UDP_RECEIVE_BUFFER,
        )

        # El socket debe ser non-blocking para asyncio.
        self.socket.setblocking(False)

        self.socket.bind(
            (
                self.host,
                self.port,
            )
        )

        self.running = True

        self.task = asyncio.create_task(
            self._run(),
            name="udp-server",
        )

        print(
            f"[UDP] Escuchando en "
            f"{self.host}:{self.port}"
        )

    # -----------------------------------------------------------------------
    # STOP
    # -----------------------------------------------------------------------

    async def stop(self):
        """
        Detiene el servidor UDP.
        """

        if not self.running:
            return

        self.running = False

        # Cerramos el socket para despertar recvfrom.
        if self.socket is not None:

            try:
                self.socket.close()
            except Exception:
                pass

            self.socket = None

        # Esperamos la task.
        if self.task is not None:

            try:
                await asyncio.wait_for(
                    self.task,
                    timeout=3.0,
                )

            except asyncio.TimeoutError:

                self.task.cancel()

                try:
                    await self.task
                except asyncio.CancelledError:
                    pass

            except asyncio.CancelledError:
                pass

        self.task = None

        print("[UDP] Servidor detenido")

    # -----------------------------------------------------------------------
    # RUN
    # -----------------------------------------------------------------------

    async def _run(self):
        """
        Loop principal del servidor UDP.
        """

        loop = asyncio.get_running_loop()

        while self.running:

            try:

                data, address = await loop.sock_recvfrom(
                    self.socket,
                    UDP_MAX_PACKET_SIZE,
                )

            except asyncio.CancelledError:

                raise

            except (OSError, RuntimeError):

                if not self.running:
                    break

                continue

            # ---------------------------------------------------------------
            # Decodificamos el paquete.
            # ---------------------------------------------------------------

            try:

                (
                    call_id,
                    metadata_json,
                    audio,
                ) = decode_udp_packet(data)

            except ValueError as exc:

                print(
                    f"[UDP] Paquete inválido desde "
                    f"{address}: {exc}"
                )

                continue

            # ---------------------------------------------------------------
            # Enviamos el paquete al manager.
            #
            # Se hace await, pero receive() no realiza I/O bloqueante.
            # ---------------------------------------------------------------

            try:

                await self.manager.receive(
                    call_id=call_id,
                    audio=audio,
                    metadata_json=metadata_json,
                )

            except Exception as exc:

                print(
                    f"[UDP] Error procesando paquete | "
                    f"call_id={call_id} | "
                    f"from={address} | "
                    f"error={exc}"
                )

# AUDIO MANAGER
# ===========================================================================

class AudioManager:
    """
    Administra el pool de TEN VAD.

    Ejemplo:

        VAD:
            127.0.0.1:12001 -> OUT 9001
            127.0.0.1:12002 -> OUT 9002
            127.0.0.1:12003 -> OUT 9003

    Primera llamada:

        call_001 -> 12001 -> 9001

    Segunda:

        call_002 -> 12002 -> 9002

    Tercera:

        call_003 -> 12003 -> 9003

    Mientras la llamada siga activa, SIEMPRE permanece en el mismo VAD.
    """

    def __init__(
        self,
        vad_servers,
        out_host,
        out_base_port,
        call_timeout=DEFAULT_CALL_TIMEOUT,
        max_pending=DEFAULT_MAX_PENDING,
    ):
        self.vad_servers = vad_servers

        self.out_host = out_host

        self.out_base_port = out_base_port

        self.call_timeout = call_timeout

        self.max_pending = max_pending

        # ---------------------------------------------------------------
        # Lock asíncrono.
        # ---------------------------------------------------------------

        self.lock = asyncio.Lock()

        # ---------------------------------------------------------------
        # Sesiones activas:
        #
        # call_id -> VadSession
        # ---------------------------------------------------------------

        self.sessions = {}

        # ---------------------------------------------------------------
        # Pool de VAD disponibles.
        #
        # Cada elemento:
        #
        #     (vad_server, out_port)
        # ---------------------------------------------------------------

        self.available = deque()

        # ---------------------------------------------------------------
        # Llamadas esperando VAD.
        #
        # Guarda:
        #
        #     call_id -> metadata
        #
        # El audio pendiente se almacena en la sesión pending.
        # ---------------------------------------------------------------

        self.pending = deque()

        self.pending_set = set()

        # ---------------------------------------------------------------
        # Task que revisa timeouts.
        # ---------------------------------------------------------------

        self.cleanup_task = None

        self.running = False

        # ---------------------------------------------------------------
        # Construimos pool.
        # ---------------------------------------------------------------

        for index, vad_server in enumerate(
            self.vad_servers
        ):

            out_port = (
                self.out_base_port + index
            )

            self.available.append(
                (
                    vad_server,
                    out_port,
                )
            )

    # -----------------------------------------------------------------------
    # START
    # -----------------------------------------------------------------------

    async def start(self):
        """
        Inicia el manager y el proceso de limpieza.
        """

        if self.running:
            return

        self.running = True

        self.cleanup_task = asyncio.create_task(
            self._cleanup_loop(),
            name="audio-manager-cleanup",
        )

        print(
            f"[MANAGER] Iniciado | "
            f"VAD disponibles={len(self.available)} | "
            f"timeout={self.call_timeout}s"
        )

    # -----------------------------------------------------------------------
    # RECEIVE
    # -----------------------------------------------------------------------

    async def receive(
        self,
        call_id,
        audio,
        metadata_json,
    ):
        """
        Recibe audio correspondiente a un call_id.

        Si la llamada ya existe:

            audio -> misma sesión

        Si es una llamada nueva:

            se asigna el primer VAD disponible.

        Si no existe VAD disponible:

            se coloca en pending.
        """

        session = None

        new_session = False

        # ===============================================================
        # PROTEGEMOS LA ASIGNACIÓN DEL POOL
        # ===============================================================

        async with self.lock:

            # -----------------------------------------------------------
            # ¿Ya existe la llamada?
            # -----------------------------------------------------------

            session = self.sessions.get(
                call_id
            )

            if session is not None:

                # La llamada ya tiene VAD.
                pass

            else:

                # -------------------------------------------------------
                # ¿Hay VAD disponible?
                # -------------------------------------------------------

                if self.available:

                    (
                        vad_server,
                        out_port,
                    ) = self.available.popleft()

                    session = VadSession(
                        call_id=call_id,
                        vad_server=vad_server,
                        out_host=self.out_host,
                        out_port=out_port,
                        call_timeout=self.call_timeout,
                    )

                    self.sessions[call_id] = session

                    new_session = True

                    print(
                        f"[MANAGER] Nueva llamada | "
                        f"call_id={call_id} | "
                        f"vad={vad_server} | "
                        f"out={self.out_host}:{out_port}"
                    )

                else:

                    # ---------------------------------------------------
                    # No hay VAD.
                    #
                    # Creamos una sesión pendiente.
                    # ---------------------------------------------------

                    # ---------------------------------------------------
                    # En este caso todavía no podemos entregar el audio
                    # a una VadSession porque todavía no existe.
                    #
                    # Por seguridad, no guardamos audio UDP ilimitadamente
                    # aquí.
                    # ---------------------------------------------------

                    return

        # ===============================================================
        # INICIAMOS LA SESIÓN FUERA DEL LOCK
        # ===============================================================

        if new_session:

            await session.start()

        # ===============================================================
        # ENTREGAMOS EL AUDIO
        # ===============================================================

        await session.push_audio(
            audio=audio,
            metadata_json=metadata_json,
        )

    # -----------------------------------------------------------------------
    # RELEASE
    # -----------------------------------------------------------------------

    async def release(
        self,
        call_id,
    ):
        """
        Libera una llamada y devuelve su VAD al pool.
        """

        session = None

        async with self.lock:

            session = self.sessions.pop(
                call_id,
                None,
            )

            if session is None:
                return

            self.available.append(
                (
                    session.vad_server,
                    session.out_port,
                )
            )

            print(
                f"[MANAGER] Liberando llamada | "
                f"call_id={call_id} | "
                f"vad={session.vad_server} | "
                f"out={session.out_port}"
            )

        # ---------------------------------------------------------------
        # Detenemos fuera del lock.
        # ---------------------------------------------------------------

        await session.stop()

        # ---------------------------------------------------------------
        # Intentamos activar una llamada pendiente.
        # ---------------------------------------------------------------

        await self._assign_pending()

    # -----------------------------------------------------------------------
    # ASSIGN PENDING
    # -----------------------------------------------------------------------

    async def _assign_pending(self):
        """
        Asigna VAD libres a llamadas pendientes.

        Ejemplo:

            pending:
                call_006
                call_007

            available:
                VAD 12003
                VAD 12004

        Resultado:

            call_006 -> 12003
            call_007 -> 12004

        IMPORTANTE:

        Como UDP no puede hacer backpressure sobre audio_headset.py,
        esta implementación solamente conserva el call_id pendiente.

        Para conservar TODO el audio mientras espera VAD habría que agregar
        un buffer por llamada, memoria compartida o almacenamiento temporal.
        """

        assignments = []

        async with self.lock:

            while (
                self.available
                and self.pending
            ):

                call_id = self.pending.popleft()

                self.pending_set.discard(
                    call_id
                )

                # -------------------------------------------------------
                # Si por alguna razón la llamada ya fue creada,
                # la ignoramos.
                # -------------------------------------------------------

                if call_id in self.sessions:
                    continue

                (
                    vad_server,
                    out_port,
                ) = self.available.popleft()

                session = VadSession(
                    call_id=call_id,
                    vad_server=vad_server,
                    out_host=self.out_host,
                    out_port=out_port,
                    call_timeout=self.call_timeout,
                )

                self.sessions[call_id] = session

                assignments.append(
                    session
                )

                print(
                    f"[MANAGER] PENDING -> VAD | "
                    f"call_id={call_id} | "
                    f"vad={vad_server} | "
                    f"out={self.out_host}:{out_port}"
                )

        # ---------------------------------------------------------------
        # Iniciamos todas las sesiones fuera del lock.
        # ---------------------------------------------------------------

        if assignments:

            await asyncio.gather(
                *(
                    session.start()
                    for session in assignments
                )
            )

    # -----------------------------------------------------------------------
    # CLEANUP LOOP
    # -----------------------------------------------------------------------

    async def _cleanup_loop(self):
        """
        Revisa periódicamente las llamadas inactivas.

        Si una llamada no recibe audio durante call_timeout:

            session -> release()

        Esto evita que un call_id quede ocupando un VAD
        indefinidamente.
        """

        try:

            while self.running:

                await asyncio.sleep(
                    DEFAULT_CLEANUP_INTERVAL
                )

                now = time.monotonic()

                expired = []

                async with self.lock:

                    for (
                        call_id,
                        session,
                    ) in list(
                        self.sessions.items()
                    ):

                        inactive_time = (
                            now - session.last_audio
                        )

                        if (
                            inactive_time
                            >= self.call_timeout
                        ):

                            expired.append(
                                call_id
                            )

                # -------------------------------------------------------
                # Liberamos fuera del lock.
                # -------------------------------------------------------

                for call_id in expired:

                    print(
                        f"[MANAGER] Timeout de llamada | "
                        f"call_id={call_id}"
                    )

                    await self.release(
                        call_id
                    )

        except asyncio.CancelledError:

            pass

    # -----------------------------------------------------------------------
    # STOP ALL
    # -----------------------------------------------------------------------

    async def stop_all(self):
        """
        Detiene todas las llamadas y libera todos los recursos.
        """

        self.running = False

        # ---------------------------------------------------------------
        # Detener cleanup.
        # ---------------------------------------------------------------

        if self.cleanup_task is not None:

            self.cleanup_task.cancel()

            try:
                await self.cleanup_task
            except asyncio.CancelledError:
                pass

            self.cleanup_task = None

        # ---------------------------------------------------------------
        # Extraemos las sesiones.
        # ---------------------------------------------------------------

        async with self.lock:

            sessions = list(
                self.sessions.values()
            )

            self.sessions.clear()

            self.pending.clear()

            self.pending_set.clear()

            # -----------------------------------------------------------
            # Reconstruimos el pool como disponible.
            # -----------------------------------------------------------

            self.available.clear()

            for index, vad_server in enumerate(
                self.vad_servers
            ):

                out_port = (
                    self.out_base_port
                    + index
                )

                self.available.append(
                    (
                        vad_server,
                        out_port,
                    )
                )

        # ---------------------------------------------------------------
        # Detenemos todas las sesiones en paralelo.
        # ---------------------------------------------------------------

        if sessions:

            await asyncio.gather(
                *(
                    session.stop()
                    for session in sessions
                ),
                return_exceptions=True,
            )

        print("[MANAGER] Todas las sesiones detenidas")


# ===========================================================================
# ARGUMENTOS
# ===========================================================================

def parse_args():
    """
    Procesa argumentos de línea de comandos.

    __init__ no puede ser async y esta función solamente procesa

    """

    parser = argparse.ArgumentParser(

        )

async def main():
    """
    Punto de entrada principal.

    Toda la aplicación corre dentro de un único event loop.
    """

    args = parse_args()

    # -----------------------------------------------------------------------
    # EVENTO DE STOP
    # -----------------------------------------------------------------------

    stop_event = asyncio.Event()

    # -----------------------------------------------------------------------
    # SIGNALS
    #
    # add_signal_handler evita tener que crear callbacks normales
    # para SIGINT/SIGTERM.
    # -----------------------------------------------------------------------

    loop = asyncio.get_running_loop()

    try:

        loop.add_signal_handler(
            signal.SIGINT,
            stop_event.set,
        )

        loop.add_signal_handler(
            signal.SIGTERM,
            stop_event.set,
        )

    except NotImplementedError:

        # Principalmente para plataformas que no soporten
        # signal handlers asyncio.
        pass

    # -----------------------------------------------------------------------
    # MANAGER
    # -----------------------------------------------------------------------

    manager = AudioManager(
        vad_servers=args.vad_servers,
        out_host=args.out_host,
        out_base_port=args.out_base_port,
        call_timeout=args.call_timeout,
        max_pending=args.max_pending,
    )

    # -----------------------------------------------------------------------
    # UDP SERVER
    # -----------------------------------------------------------------------

    udp_server = UDPServer(
        host=args.udp_host,
        port=args.udp_port,
        manager=manager,
    )

    # -----------------------------------------------------------------------
    # START
    # -----------------------------------------------------------------------

    await manager.start()

    await udp_server.start()

    # -----------------------------------------------------------------------
    # INFORMACIÓN
    # -----------------------------------------------------------------------

    print()
    print("=" * 70)
    print("ASYNC AUDIO INPUT MANAGER")
    print("=" * 70)
    print(
        f"UDP               : "
        f"{args.udp_host}:{args.udp_port}"
    )
    print(
        f"VAD disponibles    : "
        f"{len(args.vad_servers)}"
    )
    print(
        f"VAD primer puerto  : "
        f"{args.vad_servers[0]}"
    )
    print(
        f"OUT host           : "
        f"{args.out_host}"
    )
    print(
        f"OUT base port      : "
        f"{args.out_base_port}"
    )
    print(
        f"Call timeout       : "
        f"{args.call_timeout}s"
    )
    print(
        f"Max pending        : "
        f"{args.max_pending}"
    )
    print("=" * 70)
    print()

    # -----------------------------------------------------------------------
    # ESPERAMOS SIGINT / SIGTERM
    # -----------------------------------------------------------------------

    await stop_event.wait()

    print()
    print("[MAIN] Señal de apagado recibida")

    # -----------------------------------------------------------------------
    # SHUTDOWN
    # -----------------------------------------------------------------------

    await udp_server.stop()

    await manager.stop_all()

    print("[MAIN] Shutdown completo")


# ===========================================================================
# ENTRY POINT
# ===========================================================================

if __name__ == "__main__":

    try:

        asyncio.run(
            main()
        )

    except KeyboardInterrupt:

        # Normalmente SIGINT ya es manejado por asyncio,
        # pero dejamos este fallback.
        pass

def encode_udp_packet(
    call_id,
    metadata_json,
    audio,
):
    header = {
        "call_id": call_id,
        "metadata_json": metadata_json,
    }

    header_bytes = json.dumps(
        header,
        separators=(",", ":"),
    ).encode("utf-8")

    return (
        struct.pack(
            "!I",
            len(header_bytes),
        )
        + header_bytes
        + audio
    )


class UDPAudioSender:

    def __init__(
        self,
        host,
        port,
        copies,
        call_id_prefix,
    ):

        self.address = (
            host,
            port,
        )

        self.socket = socket.socket(
            socket.AF_INET,
            socket.SOCK_DGRAM,
        )

        self.targets = []

        for index in range(copies):

            call_id = (
                f"{call_id_prefix}-"
                f"{index + 1:03d}"
            )

            metadata_json = json.dumps(
                {
                    "cliente_id": (
                        f"12345-{index + 1}"
                    ),
                    "campana": "ventas",
                    "agente": (
                        f"agente-{index + 1:03d}"
                    ),

                    # El manager reemplazará
                    # estos valores con el
                    # OUT correspondiente.
                    "out_host": "",
                    "out_port": 0,
                },
                separators=(",", ":"),
            )

            self.targets.append(
                (
                    call_id,
                    metadata_json,
                )
            )

            print(
                f"[HEADSET] "
                f"{call_id} -> "
                f"UDP {host}:{port}"
            )

    def send(
        self,
        audio,
    ):

        for call_id, metadata_json in self.targets:

            packet = encode_udp_packet(
                call_id=call_id,
                metadata_json=metadata_json,
                audio=audio,
            )

            try:

                self.socket.sendto(
                    packet,
                    self.address,
                )

            except OSError as exc:

                print(
                    f"[HEADSET] ERROR "
                    f"{call_id}: {exc}",
                    file=sys.stderr,
                )

    def close(self):

        self.socket.close()

class SegmentOutputService(
    audio_listener_pb2_grpc.SegmentOutputServicer
):

    def __init__(
        self,
        output_directory,
    ):

        self.output_directory = Path(
            output_directory
        )

        self.output_directory.mkdir(
            parents=True,
            exist_ok=True,
        )

        self.lock = threading.Lock()

        print(
            f"[OUT] Audio directory: "
            f"{self.output_directory}"
        )

    def SendSegment(
        self,
        request,
        context,
    ):

        call_id = request.call_id
        segment_id = request.segment_id

        if not call_id:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "call_id es obligatorio"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        if not request.audio:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "audio es obligatorio"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        sample_rate = (
            request.sample_rate
            or 16000
        )

        channels = (
            request.channels
            or 1
        )

        if sample_rate != 16000:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "Solo se acepta audio a 16000 Hz"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        if channels != 1:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "Solo se acepta audio mono"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        audio_bytes = request.audio

        if len(audio_bytes) % 2 != 0:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "Audio PCM16 inválido"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        duration = (
            (len(audio_bytes) / 2)
            / sample_rate
        )

        call_directory = (
            self.output_directory /
            call_id
        )

        call_directory.mkdir(
            parents=True,
            exist_ok=True,
        )

        output_file = (
            call_directory /
            f"{segment_id}.wav"
        )

        try:

            with self.lock:

                with wave.open(
                    str(output_file),
                    "wb",
                ) as wav:

                    wav.setnchannels(
                        channels
                    )

                    wav.setsampwidth(
                        2
                    )

                    wav.setframerate(
                        sample_rate
                    )

                    wav.writeframes(
                        audio_bytes
                    )

        except Exception as exc:

            print(
                f"[OUT] ERROR guardando "
                f"{output_file}: {exc}",
                file=sys.stderr,
            )

            context.set_code(
                grpc.StatusCode.INTERNAL
            )

            context.set_details(
                "No se pudo guardar WAV"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        print(
            f"[OUT] "
            f"call={call_id} | "
            f"segment={segment_id} | "
            f"{duration:.3f}s | "
            f"{len(audio_bytes)} bytes | "
            f"{output_file}"
        )

        return audio_listener_pb2.SegmentResponse(
            success=True
        )

    def Ping(
        self,
        request,
        context,
    ):

        print(
            f"[OUT] Ping received | "
            f"source={request.source}"
        )

        return audio_listener_pb2.PingResponse(
            success=True,
            service="audio-out",
            message="OUT is alive",
        )
