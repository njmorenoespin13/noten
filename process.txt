async def convert_response_to_words(self, response: str) -> str:
    """
    Convierte los números de una respuesta final a palabras en español.

    No ejecuta herramientas ni modifica el resultado de los cálculos.
    """

    if not response:
        return response

    conversion_prompt = """
Eres un conversor de respuestas numéricas a palabras en español.

Tu única tarea es reescribir el texto recibido convirtiendo sus números
a palabras, conservando exactamente su significado.

REGLAS OBLIGATORIAS:

1. Convierte los valores monetarios a palabras completas en español.
   Ejemplo: $200.000 -> doscientos mil pesos.
   Ejemplo: $1.250.000 -> un millón doscientos cincuenta mil pesos.

2. Convierte las fechas completas a palabras.
   Ejemplo: 13 de octubre de 2026 ->
   trece de octubre de dos mil veintiséis.

3. Convierte cantidades, días, meses, años y porcentajes a palabras.
   Ejemplo: 15 días -> quince días.
   Ejemplo: 20% -> veinte por ciento.

4. Interpreta los puntos como separadores de miles cuando corresponda
   al formato monetario colombiano.

5. No conviertas cada dígito individualmente.
   Debes expresar el número completo correctamente en español.
   Por ejemplo, 100.000 es "cien mil", no "uno, cero, cero, cero, cero, cero".

6. Conserva el significado, el tono, los nombres y toda la información
   del texto original.

7. No inventes datos, operaciones, cantidades ni fechas.
   No recalcules ni modifiques ningún resultado.

8. No agregues explicaciones, saludos, títulos ni comentarios.

9. Devuelve exclusivamente el texto final transformado.

10. Si aparece un código, teléfono o identificador, consérvalo sin cambios.

TEXTO ORIGINAL:
"""

    try:
        result = await self.llm.ainvoke(
            [
                SystemMessage(content=conversion_prompt),
                HumanMessage(content=response),
            ]
        )

        content = result.content

        if isinstance(content, str) and content.strip():
            return content.strip()

        return response

    except Exception:
        logging.exception(
            "Error convirtiendo la respuesta final a palabras"
        )

        # Si falla la conversión, conservamos la respuesta original.
        return response


async def extract_final_response(
    self,
    state: AgentState,
) -> str:

    messages = state["messages"]

    # ----------------------------------------------------
    # Primero buscamos herramientas terminales.
    # ----------------------------------------------------

    terminal_messages = [
        message
        for message in messages
        if (
            isinstance(message, ToolMessage)
            and message.name in TERMINAL_TOOLS
        )
    ]

    if terminal_messages:

        responses = []

        for message in terminal_messages:

            formatted = ResponseFormatter.format(
                message.name,
                message.content,
            )

            responses.append(formatted)

        response = "\n".join(responses)

        # Segunda llamada al modelo:
        # solamente convierte la respuesta final a palabras.
        return await self.convert_response_to_words(response)

    # ----------------------------------------------------
    # Si no hubo herramientas, usamos la respuesta del LLM.
    # ----------------------------------------------------

    for message in reversed(messages):

        if isinstance(message, AIMessage):

            content = message.content

            if isinstance(content, str):

                response = content.strip()

                return await self.convert_response_to_words(
                    response
                )

            if isinstance(content, list):

                text_parts = []

                for item in content:

                    if isinstance(item, dict):

                        text = item.get("text")

                        if text:
                            text_parts.append(text)

                if text_parts:

                    response = "".join(text_parts).strip()

                    return await self.convert_response_to_words(
                        response
                    )

    return ""
