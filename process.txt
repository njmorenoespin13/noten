
#!/bin/bash

set -u

# ==========================================
# CONFIGURACION
# ==========================================

GITEA_URL="http://127.0.0.1:3000"
REPO_ROOT="/var/lib/gitea/git/repositories"

DEFAULT_PASSWORD="contraseña"
DEFAULT_EMAIL_DOMAIN="mail.com"

SUCCESS_USERS=0
EXISTING_USERS=0
ERROR_USERS=0

SUCCESS_REPOS=0
EXISTING_REPOS=0
ERROR_REPOS=0

LOG="/var/log/gitea_importacion_$(date +%Y%m%d_%H%M%S).log"
TMP=$(mktemp)

trap 'rm -f "$TMP"' EXIT

# ==========================================
# FUNCIONES
# ==========================================

log() {
    local MESSAGE="$1"
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $MESSAGE" | tee -a "$LOG"
}

urlencode() {
    jq -rn --arg value "$1" '$value | @uri'
}

api_request() {
    local METHOD="$1"
    local URL="$2"
    local DATA="${3:-}"

    if [[ -n "$DATA" ]]; then
        curl -sS \
            --connect-timeout 10 \
            --max-time 60 \
            -o "$TMP" \
            -w "%{http_code}" \
            -X "$METHOD" \
            -H "Authorization: token $TOKEN" \
            -H "Content-Type: application/json" \
            --data "$DATA" \
            "$URL"
    else
        curl -sS \
            --connect-timeout 10 \
            --max-time 60 \
            -o "$TMP" \
            -w "%{http_code}" \
            -X "$METHOD" \
            -H "Authorization: token $TOKEN" \
            "$URL"
    fi
}

# ==========================================
# VALIDACIONES
# ==========================================

if [[ ! -d "$REPO_ROOT" ]]; then
    echo "ERROR: No existe la ruta $REPO_ROOT"
    exit 1
fi

for CMD in curl jq find; do
    if ! command -v "$CMD" >/dev/null 2>&1; then
        echo "ERROR: Falta instalar $CMD"
        exit 1
    fi
done

echo
echo "=========================================="
echo "       IMPORTADOR MASIVO GITEA"
echo "=========================================="
echo "Ruta: $REPO_ROOT"
echo "Servidor: $GITEA_URL"
echo

read -rsp "Token del administrador Gitea: " TOKEN
echo
echo

# Comprobar autenticacion
HTTP=$(api_request GET "$GITEA_URL/api/v1/user")

if [[ "$HTTP" != "200" ]]; then
    echo "ERROR: No se pudo autenticar con Gitea. HTTP $HTTP"
    cat "$TMP"
    exit 1
fi

log "Conexion con Gitea correcta."

# Contar usuarios y repositorios
USER_COUNT=0
REPO_COUNT=0

for OWNER_DIR in "$REPO_ROOT"/*/; do
    [[ -d "$OWNER_DIR" ]] || continue

    ((USER_COUNT+=1))

    while IFS= read -r -d '' REPO_DIR; do
        ((REPO_COUNT+=1))
    done < <(find "$OWNER_DIR" -mindepth 1 -maxdepth 1 \
        -type d -name "*.git" -print0)
done

echo
echo "Usuarios detectados: $USER_COUNT"
echo "Repositorios detectados: $REPO_COUNT"
echo
read -rp "¿Deseas iniciar la importación? (s/N): " CONFIRM

if [[ ! "$CONFIRM" =~ ^[sS]$ ]]; then
    echo "Proceso cancelado."
    exit 0
fi

# ==========================================
# PROCESAMIENTO
# ==========================================

for OWNER_DIR in "$REPO_ROOT"/*/; do

    [[ -d "$OWNER_DIR" ]] || continue

    OWNER=$(basename "$OWNER_DIR")
    OWNER_ENC=$(urlencode "$OWNER")

    echo
    log "------------------------------------------"
    log "PROCESANDO USUARIO: $OWNER"

    # --------------------------------------
    # COMPROBAR USUARIO
    # --------------------------------------

    HTTP=$(api_request GET \
        "$GITEA_URL/api/v1/users/$OWNER_ENC")

    if [[ "$HTTP" == "200" ]]; then

        log "[USUARIO EXISTENTE] $OWNER - No se modifica"
        ((EXISTING_USERS+=1))

    elif [[ "$HTTP" == "404" ]]; then

        EMAIL="${OWNER}@${DEFAULT_EMAIL_DOMAIN}"

        USER_DATA=$(jq -n \
            --arg username "$OWNER" \
            --arg email "$EMAIL" \
            --arg password "$DEFAULT_PASSWORD" \
            '{
                username: $username,
                email: $email,
                password: $password,
                must_change_password: true
            }')

        HTTP=$(api_request POST \
            "$GITEA_URL/api/v1/admin/users" \
            "$USER_DATA")

        if [[ "$HTTP" == "201" ]]; then

            log "[USUARIO CREADO] $OWNER ($EMAIL)"
            ((SUCCESS_USERS+=1))

        else

            log "[ERROR USUARIO] $OWNER - HTTP $HTTP"
            log "Respuesta: $(cat "$TMP")"
            ((ERROR_USERS+=1))
            continue

        fi

    else

        log "[ERROR USUARIO] $OWNER - HTTP $HTTP"
        log "Respuesta: $(cat "$TMP")"
        ((ERROR_USERS+=1))
        continue

    fi

    # --------------------------------------
    # PROCESAR REPOSITORIOS
    # --------------------------------------

    while IFS= read -r -d '' REPO_DIR; do

        REPO_FOLDER=$(basename "$REPO_DIR")
        REPO="${REPO_FOLDER%.git}"
        REPO_ENC=$(urlencode "$REPO")

        log "Repositorio: $OWNER/$REPO"

        # Comprobar si ya existe en Gitea
        HTTP=$(api_request GET \
            "$GITEA_URL/api/v1/repos/$OWNER_ENC/$REPO_ENC")

        if [[ "$HTTP" == "200" ]]; then

            log "[REPOSITORIO EXISTENTE] $OWNER/$REPO - Sin cambios"
            ((EXISTING_REPOS+=1))
            continue

        elif [[ "$HTTP" != "404" ]]; then

            log "[ERROR CONSULTA] $OWNER/$REPO - HTTP $HTTP"
            log "Respuesta: $(cat "$TMP")"
            ((ERROR_REPOS+=1))
            continue

        fi

        # Adoptar repositorio existente en disco
        HTTP=$(api_request POST \
            "$GITEA_URL/api/v1/admin/unadopted/$OWNER_ENC/$REPO_ENC")

        if [[ "$HTTP" =~ ^2[0-9][0-9]$ ]]; then

            log "[REPOSITORIO ADOPTADO] $OWNER/$REPO"
            ((SUCCESS_REPOS+=1))

        else

            log "[ERROR ADOPCION] $OWNER/$REPO - HTTP $HTTP"
            log "Respuesta: $(cat "$TMP")"
            ((ERROR_REPOS+=1))

        fi

    done < <(find "$OWNER_DIR" -mindepth 1 -maxdepth 1 \
        -type d -name "*.git" -print0)

done

# ==========================================
# RESUMEN
# ==========================================

echo
echo "=========================================="
echo "             RESUMEN FINAL"
echo "=========================================="

echo "Usuarios creados:       $SUCCESS_USERS"
echo "Usuarios existentes:    $EXISTING_USERS"
echo "Errores de usuarios:    $ERROR_USERS"

echo "Repositorios adoptados: $SUCCESS_REPOS"
echo "Repositorios existentes:$EXISTING_REPOS"
echo "Errores de repositorios:$ERROR_REPOS"

echo
echo "Log: $LOG"
echo "=========================================="
