# ============================================================
# UDP PACKET
# ============================================================
#
# El headset NO manda directamente gRPC.
#
# Primero manda un paquete UDP al único puerto del Manager.
#
# Ejemplo:
#
# audio_headset.py
#        |
#        | UDP :12000
#        v
# audio_in_manager.py
#
# El paquete UDP tiene esta estructura:
#
# +----------------------+-----------------------+----------------+
# | 4 bytes              | Header JSON           | Audio PCM      |
# | tamaño del header    |                       |                |
# +----------------------+-----------------------+----------------+
#
# Los primeros 4 bytes indican cuánto mide el JSON.
#
# Ejemplo conceptual:
#
# [00 00 00 C8] [JSON.....................] [audio...........]
#
# Esto permite saber exactamente dónde termina el JSON
# y dónde comienza el audio.
# ============================================================

def decode_udp_packet(data):
    """
    Decodifica un paquete UDP recibido.

    Devuelve:

        call_id
        metadata_json
        audio
    """

    # --------------------------------------------------------
    # Verificamos que al menos existan los 4 bytes del header.
    # --------------------------------------------------------

    if len(data) < 4:
        raise ValueError(
            "Paquete UDP demasiado pequeño"
        )


    # --------------------------------------------------------
    # Leemos los primeros 4 bytes.
    #
    # "!I" significa:
    #
    # !  -> network byte order (big endian)
    # I  -> unsigned int de 4 bytes
    #
    # Esto nos da el tamaño del JSON.
    # --------------------------------------------------------

    header_size = struct.unpack(
        "!I",
        data[:4],
    )[0]


    # --------------------------------------------------------
    # Validamos que el tamaño indicado sea correcto.
    # --------------------------------------------------------

    if (
        header_size <= 0
        or
        len(data) < 4 + header_size
    ):
        raise ValueError(
            "Header UDP inválido"
        )


    # --------------------------------------------------------
    # Extraemos solamente la parte JSON.
    #
    # data[4:] comienza después de los 4 bytes iniciales.
    # --------------------------------------------------------

    header_bytes = data[
        4:
        4 + header_size
    ]


    # --------------------------------------------------------
    # Todo lo que viene después del JSON es audio.
    #
    # En nuestro caso normalmente será PCM:
    #
    # 16 kHz
    # mono
    # S16_LE
    # --------------------------------------------------------

    audio = data[
        4 + header_size:
    ]


    # --------------------------------------------------------
    # Convertimos el JSON de bytes -> string -> diccionario.
    # --------------------------------------------------------

    header = json.loads(
        header_bytes.decode("utf-8")
    )


    # --------------------------------------------------------
    # Obtenemos el identificador de llamada.
    #
    # Este valor es MUY importante porque es el que permite
    # separar las llamadas.
    #
    # Ejemplo:
    #
    # call-001
    # call-002
    # call-003
    #
    # Todas pueden llegar por el mismo UDP :12000.
    # --------------------------------------------------------

    call_id = header.get(
        "call_id"
    )


    # --------------------------------------------------------
    # metadata_json contiene información adicional.
    #
    # Por ejemplo:
    #
    # cliente_id
    # campana
    # agente
    # etc.
    #
    # Si no viene, utilizamos "".
    # --------------------------------------------------------

    metadata_json = header.get(
        "metadata_json",
        "",
    )


    # --------------------------------------------------------
    # Validaciones básicas.
    # --------------------------------------------------------

    if not call_id:
        raise ValueError(
            "call_id faltante"
        )

    if not audio:
        raise ValueError(
            "audio vacío"
        )


    # --------------------------------------------------------
    # Devolvemos los tres elementos separados.
    # --------------------------------------------------------

    return (
        call_id,
        metadata_json,
        audio,
    )


# ============================================================
# GRPC VAD SESSION
# ============================================================
#
# Esta clase representa UNA llamada conectada a UN TEN VAD.
#
# Ejemplo:
#
# call-001
#     |
#     +--> VAD 127.0.0.1:12001
#              |
#              +--> OUT :9001
#
#
# call-002
#     |
#     +--> VAD 127.0.0.1:12002
#              |
#              +--> OUT :9002
#
# Cada sesión tiene:
#
# - Su propio queue.Queue()
# - Su propio canal gRPC
# - Su propio thread
# - Su propio call_id
# - Su propio TEN VAD
# - Su propio OUT
#
# Esto evita que el audio de una llamada se mezcle con otra.
# ============================================================

class VadSession:

    def __init__(
        self,
        call_id,
        vad_server,
        out_host,
        out_port,
    ):

        # ----------------------------------------------------
        # Identificador de la llamada.
        # ----------------------------------------------------

        self.call_id = call_id


        # ----------------------------------------------------
        # Dirección del TEN VAD asignado.
        #
        # Ejemplo:
        #
        # 127.0.0.1:12001
        # ----------------------------------------------------

        self.vad_server = vad_server


        # ----------------------------------------------------
        # Dirección del OUT asociado a este TEN VAD.
        #
        # Ejemplo:
        #
        # host.docker.internal:9001
        # ----------------------------------------------------

        self.out_host = out_host
        self.out_port = out_port


        # ----------------------------------------------------
        # Cola exclusiva para esta llamada.
        #
        # El UDP Manager mete aquí los audios.
        #
        # Otro thread consume estos audios y los manda
        # mediante gRPC al TEN VAD.
        #
        # IMPORTANTE:
        #
        # No existe una cola global.
        #
        # Cada llamada tiene su propia cola.
        # ----------------------------------------------------

        self.queue = queue.Queue()


        # ----------------------------------------------------
        # Canal gRPC hacia TEN VAD.
        # ----------------------------------------------------

        self.channel = None


        # Stub que nos permite llamar al servicio TenVad.
        self.stub = None


        # Thread que mantiene la comunicación con TEN VAD.
        self.thread = None


        # Indica si esta sesión continúa activa.
        self.running = True


        # Último momento en que recibimos audio.
        #
        # Se utiliza posteriormente para detectar llamadas
        # que dejaron de enviar audio.
        self.last_audio = time.monotonic()


        # Último metadata recibido.
        self.metadata_json = ""


    # ========================================================
    # START
    # ========================================================

    def start(self):
        """
        Inicia el thread encargado de conectar con TEN VAD.

        Importante:

        El UDP Server NO se queda esperando a que gRPC
        conecte.

        La conexión gRPC ocurre en otro thread.
        """

        self.thread = threading.Thread(
            target=self._run,
            name=f"vad-{self.call_id}",
            daemon=True,
        )

        self.thread.start()


    # ========================================================
    # PUSH AUDIO
    # ========================================================

    def push_audio(
        self,
        audio,
        metadata_json,
    ):
        """
        Recibe audio desde el UDP Manager y lo mete
        en la cola de ESTA llamada.
        """

        # ----------------------------------------------------
        # Actualizamos el último momento en que recibimos
        # audio de esta llamada.
        # ----------------------------------------------------

        self.last_audio = time.monotonic()


        # ----------------------------------------------------
        # El Manager es quien decide a qué OUT debe ir
        # esta llamada.
        #
        # NO confiamos en out_host/out_port enviados por
        # el headset.
        # ----------------------------------------------------

        try:
            metadata = json.loads(
                metadata_json
            )

        except Exception:
            # Si el metadata no es JSON válido,
            # utilizamos un diccionario vacío.
            metadata = {}


        # ----------------------------------------------------
        # Sobrescribimos el OUT con el OUT que corresponde
        # al TEN VAD asignado.
        #
        # Ejemplo:
        #
        # VAD :12001 -> OUT :9001
        #
        # VAD :12002 -> OUT :9002
        # ----------------------------------------------------

        metadata[
            "out_host"
        ] = self.out_host

        metadata[
            "out_port"
        ] = self.out_port


        # ----------------------------------------------------
        # Convertimos nuevamente el diccionario a JSON.
        #
        # separators elimina espacios innecesarios y hace
        # el mensaje un poco más pequeño.
        # ----------------------------------------------------

        self.metadata_json = json.dumps(
            metadata,
            separators=(",", ":"),
        )


        # ----------------------------------------------------
        # Guardamos audio + metadata en la cola.
        #
        # Este put() normalmente es muy rápido.
        #
        # El thread de gRPC será quien saque posteriormente
        # estos datos de la cola.
        # ----------------------------------------------------

        self.queue.put(
            (
                audio,
                self.metadata_json,
            )
        )


    # ========================================================
    # STOP
    # ========================================================

    def stop(self):
        """
        Detiene esta sesión.
        """

        # ----------------------------------------------------
        # Indicamos al generator que debe terminar.
        # ----------------------------------------------------

        self.running = False


        # ----------------------------------------------------
        # Insertamos None como señal especial.
        #
        # El generator detectará None y terminará.
        # ----------------------------------------------------

        self.queue.put(
            None
        )


        # ----------------------------------------------------
        # Esperamos máximo 5 segundos al thread.
        # ----------------------------------------------------

        if self.thread:

            self.thread.join(
                timeout=5
            )


        # ----------------------------------------------------
        # Cerramos el canal gRPC.
        # ----------------------------------------------------

        if self.channel:

            self.channel.close()


    # ========================================================
    # GENERATOR gRPC
    # ========================================================
    #
    # gRPC StreamAudio necesita un iterable/generator.
    #
    # Este generator convierte:
    #
    #     queue.Queue()
    #
    # en:
    #
    #     AudioStreamRequest
    #
    # Por cada audio que entra en la cola se genera
    # una solicitud protobuf.
    # ========================================================

    def _generator(self):

        while self.running:

            # ------------------------------------------------
            # Esperamos hasta que exista audio.
            # ------------------------------------------------

            item = self.queue.get()


            # ------------------------------------------------
            # None significa:
            #
            # "terminar la sesión"
            # ------------------------------------------------

            if item is None:
                break


            # ------------------------------------------------
            # Extraemos audio + metadata.
            # ------------------------------------------------

            audio, metadata_json = item


            # ------------------------------------------------
            # Construimos el mensaje protobuf.
            #
            # Este mensaje viaja por gRPC hacia TEN VAD.
            # ------------------------------------------------

            yield (
                audio_listener_pb2.AudioStreamRequest(
                    call_id=self.call_id,
                    audio=audio,
                    metadata_json=metadata_json,
                )
            )


    # ========================================================
    # RUN
    # ========================================================

    def _run(self):

        print(
            f"[MANAGER] "
            f"{self.call_id} → "
            f"VAD {self.vad_server} → "
            f"OUT {self.out_host}:{self.out_port}"
        )


        try:

            # ------------------------------------------------
            # Creamos el canal gRPC.
            #
            # Ejemplo:
            #
            # 127.0.0.1:12001
            # ------------------------------------------------

            self.channel = grpc.insecure_channel(
                self.vad_server
            )


            # ------------------------------------------------
            # Esperamos hasta 10 segundos a que TEN VAD
            # esté disponible.
            #
            # Esto ocurre en ESTE thread, no en el thread
            # que recibe UDP.
            # ------------------------------------------------

            grpc.channel_ready_future(
                self.channel
            ).result(
                timeout=10
            )


            # ------------------------------------------------
            # Creamos el Stub para poder llamar:
            #
            # TenVad.StreamAudio()
            # ------------------------------------------------

            self.stub = (
                audio_listener_pb2_grpc.TenVadStub(
                    self.channel
                )
            )


            # ------------------------------------------------
            # Abrimos el stream bidireccional hacia TEN VAD.
            #
            # El generator irá entregando audio conforme
            # vaya llegando a la cola.
            # ------------------------------------------------

            responses = self.stub.StreamAudio(
                self._generator()
            )


            # ------------------------------------------------
            # Esperamos las respuestas de TEN VAD.
            #
            # Actualmente TEN VAD responde success=true
            # mientras la comunicación sea correcta.
            # ------------------------------------------------

            for response in responses:

                if not response.success:

                    print(
                        f"[MANAGER] "
                        f"{self.call_id}: "
                        f"TEN VAD respondió "
                        f"success=false",
                        file=sys.stderr,
                    )


                # ------------------------------------------------
                # Si la sesión fue detenida, salimos.
                # ------------------------------------------------

                if not self.running:
                    break


        except grpc.RpcError as exc:

            # ------------------------------------------------
            # Error específico de gRPC.
            #
            # Ejemplo:
            #
            # TEN VAD apagado
            # conexión perdida
            # servidor no disponible
            # ------------------------------------------------

            if self.running:

                print(
                    f"[MANAGER] "
                    f"{self.call_id} "
                    f"gRPC ERROR: "
                    f"{exc.code()} - "
                    f"{exc.details()}",
                    file=sys.stderr,
                )


        except Exception as exc:

            # ------------------------------------------------
            # Cualquier otro error.
            # ------------------------------------------------

            print(
                f"[MANAGER] "
                f"{self.call_id} "
                f"ERROR: {exc}",
                file=sys.stderr,
            )


        finally:

            # ------------------------------------------------
            # Cuando termina la sesión cerramos el canal.
            # ------------------------------------------------

            if self.channel:

                self.channel.close()


            print(
                f"[MANAGER] "
                f"{self.call_id} "
                f"VAD session finalizada"
            )


# ============================================================
# UDP SERVER
# ============================================================
#
# Esta clase tiene UNA responsabilidad:
#
#     Recibir paquetes UDP.
#
# NO procesa audio.
# NO se conecta directamente a TEN VAD.
#
# Su trabajo es:
#
#     UDP
#       |
#       v
#   decode packet
#       |
#       v
#   AudioManager.receive()
#
# ============================================================

class UDPServer:

    def __init__(
        self,
        host,
        port,
        manager,
    ):

        self.host = host
        self.port = port
        self.manager = manager

        self.socket = None
        self.thread = None
        self.running = False


    # ========================================================
    # START UDP
    # ========================================================

    def start(self):

        # ----------------------------------------------------
        # Creamos socket UDP IPv4.
        # ----------------------------------------------------

        self.socket = socket.socket(
            socket.AF_INET,
            socket.SOCK_DGRAM,
        )


        # ----------------------------------------------------
        # Aumentamos el buffer de recepción del sistema.
        #
        # Esto ayuda cuando llegan muchos paquetes UDP
        # rápidamente.
        #
        # IMPORTANTE:
        #
        # UDP sigue siendo UDP.
        # No garantiza entrega.
        # ----------------------------------------------------

        self.socket.setsockopt(
            socket.SOL_SOCKET,
            socket.SO_RCVBUF,
            16 * 1024 * 1024,
        )


        # ----------------------------------------------------
        # Escuchamos en:
        #
        # 0.0.0.0:12000
        #
        # 0.0.0.0 significa todas las interfaces.
        # ----------------------------------------------------

        self.socket.bind(
            (
                self.host,
                self.port,
            )
        )


        self.running = True


        # ----------------------------------------------------
        # El listener UDP corre en un thread independiente.
        # ----------------------------------------------------

        self.thread = threading.Thread(
            target=self._run,
            daemon=True,
        )

        self.thread.start()


        print(
            f"[UDP] Listening on "
            f"{self.host}:{self.port}"
        )


    # ========================================================
    # STOP UDP
    # ========================================================

    def stop(self):

        self.running = False


        # ----------------------------------------------------
        # Cerramos el socket.
        #
        # Esto hace que recvfrom() deje de esperar.
        # ----------------------------------------------------

        if self.socket:

            try:
                self.socket.close()

            except Exception:
                pass


        if self.thread:

            self.thread.join(
                timeout=2
            )


    # ========================================================
    # UDP RECEIVE LOOP
    # ========================================================

    def _run(self):

        while self.running:

            try:

                # ------------------------------------------------
                # Esperamos un datagrama UDP.
                #
                # data    = contenido
                # address = IP + puerto del remitente
                # ------------------------------------------------

                data, address = (
                    self.socket.recvfrom(
                        UDP_BUFFER_SIZE
                    )
                )


            except OSError:

                # ------------------------------------------------
                # Normalmente ocurre cuando cerramos el socket.
                # ------------------------------------------------

                break


            except Exception as exc:

                print(
                    f"[UDP] ERROR: {exc}",
                    file=sys.stderr,
                )

                continue


            try:

                # ------------------------------------------------
                # Decodificamos el paquete.
                # ------------------------------------------------

                (
                    call_id,
                    metadata_json,
                    audio,
                ) = decode_udp_packet(
                    data
                )


            except Exception as exc:

                # ------------------------------------------------
                # Si el paquete está corrupto o incompleto,
                # no afecta al resto de llamadas.
                # ------------------------------------------------

                print(
                    f"[UDP] Invalid packet "
                    f"from {address}: "
                    f"{exc}",
                    file=sys.stderr,
                )

                continue


            # ----------------------------------------------------
            # Entregamos el audio al AudioManager.
            #
            # A partir de aquí AudioManager decide:
            #
            # ¿Es una llamada nueva?
            #
            # ¿Ya existe?
            #
            # ¿Qué TEN VAD le corresponde?
            # ----------------------------------------------------

            self.manager.receive(
                call_id=call_id,
                metadata_json=metadata_json,
                audio=audio,
            )


# ============================================================
# AUDIO MANAGER
# ============================================================
#
# ESTA ES LA PARTE MÁS IMPORTANTE.
#
# AudioManager administra el POOL de TEN VAD.
#
# Tenemos:
#
# UDP :12000
#       |
#       +---- call-001
#       +---- call-002
#       +---- call-003
#       +---- call-004
#
#
# Y tenemos un pool:
#
# VAD 12001 -> OUT 9001
# VAD 12002 -> OUT 9002
# VAD 12003 -> OUT 9003
# ...
#
#
# Cuando aparece una llamada NUEVA:
#
# call-001 -> primer VAD disponible
# call-002 -> segundo VAD disponible
# call-003 -> tercer VAD disponible
#
#
# Cuando llega otro paquete de call-001:
#
# call-001 -> ya existe
#          -> utiliza EL MISMO VAD
#
# ============================================================

class AudioManager:

    def __init__(
        self,
        vad_servers,
        out_host,
        out_base_port,
    ):

        # Lista de TEN VAD disponibles.
        self.vad_servers = vad_servers


        # Host donde están los OUT.
        self.out_host = out_host


        # Puerto inicial del OUT.
        #
        # Si es 9001:
        #
        # VAD #1 -> 9001
        # VAD #2 -> 9002
        # VAD #3 -> 9003
        # --------------------------------------------------------

        self.out_base_port = out_base_port


        # --------------------------------------------------------
        # Lock para proteger las estructuras compartidas.
        #
        # Como tenemos múltiples threads, dos llamadas podrían
        # intentar utilizar el pool al mismo tiempo.
        #
        # El lock evita que dos threads asignen accidentalmente
        # el mismo VAD.
        # --------------------------------------------------------

        self.lock = threading.Lock()


        # --------------------------------------------------------
        # Diccionario:
        #
        # call_id -> VadSession
        #
        # Ejemplo:
        #
        # {
        #     "call-001": sesión VAD 12001,
        #     "call-002": sesión VAD 12002
        # }
        # --------------------------------------------------------

        self.sessions = {}


        # --------------------------------------------------------
        # Lista de VAD libres.
        #
        # Al iniciar:
        #
        # [
        #   ("127.0.0.1:12001", 9001),
        #   ("127.0.0.1:12002", 9002),
        #   ...
        # ]
        # --------------------------------------------------------

        self.available = []


        # --------------------------------------------------------
        # Construimos el pool.
        #
        # enumerate() nos da:
        #
        # index = posición
        # vad_server = dirección TEN VAD
        # --------------------------------------------------------

        for index, vad_server in enumerate(
            vad_servers
        ):

            # El OUT se calcula según la posición.
            #
            # index 0 -> 9001
            # index 1 -> 9002
            # index 2 -> 9003
            out_port = (
                out_base_port +
                index
            )


            # Guardamos VAD + OUT como una pareja.
            self.available.append(
                (
                    vad_server,
                    out_port,
                )
            )


        # --------------------------------------------------------
        # Mostramos cuántos TEN VAD tenemos.
        # --------------------------------------------------------

        print(
            f"[MANAGER] "
            f"VAD services disponibles: "
            f"{len(self.available)}"
        )


        # --------------------------------------------------------
        # Mostramos el mapeo.
        #
        # Ejemplo:
        #
        # 127.0.0.1:12001 -> OUT host.docker.internal:9001
        # 127.0.0.1:12002 -> OUT host.docker.internal:9002
        # --------------------------------------------------------

        for vad_server, out_port in self.available:

            print(
                f"[MANAGER] "
                f"{vad_server} → "
                f"OUT {out_host}:{out_port}"
            )


    # ========================================================
    # RECEIVE
    # ========================================================

    def receive(
        self,
        call_id,
        metadata_json,
        audio,
    ):

        # ----------------------------------------------------
        # Protegemos la asignación del pool.
        # ----------------------------------------------------

        with self.lock:

            # ------------------------------------------------
            # Buscamos si esta llamada ya tiene una sesión.
            # ------------------------------------------------

            session = self.sessions.get(
                call_id
            )


            # =================================================
            # CASO 1:
            # LA LLAMADA ES NUEVA
            # =================================================

            if session is None:

                # ------------------------------------------------
                # Si no quedan TEN VAD libres, actualmente
                # rechazamos el audio.
                #
                # Más adelante esto puede cambiarse por una
                # cola de llamadas pendientes.
                # ------------------------------------------------

                if not self.available:

                    print(
                        f"[MANAGER] "
                        f"NO HAY VAD DISPONIBLE "
                        f"para {call_id}",
                        file=sys.stderr,
                    )

                    return


                # ------------------------------------------------
                # Sacamos el PRIMER VAD disponible.
                #
                # pop(0) mantiene el orden:
                #
                # primero -> VAD 12001
                # segundo -> VAD 12002
                # tercero -> VAD 12003
                # ------------------------------------------------

                (
                    vad_server,
                    out_port,
                ) = self.available.pop(
                    0
                )


                print(
                    f"[MANAGER] ASIGNANDO "
                    f"{call_id} → "
                    f"{vad_server} → "
                    f"OUT {self.out_host}:{out_port}"
                )


                # ------------------------------------------------
                # Creamos una sesión exclusiva para esta llamada.
                # ------------------------------------------------

                session = VadSession(
                    call_id=call_id,
                    vad_server=vad_server,
                    out_host=self.out_host,
                    out_port=out_port,
                )


                # ------------------------------------------------
                # Guardamos la relación:
                #
                # call_id -> sesión
                #
                # Desde este momento todos los siguientes
                # paquetes de call_id utilizarán ESTA sesión.
                # ------------------------------------------------

                self.sessions[
                    call_id
                ] = session


                # ------------------------------------------------
                # Arrancamos el thread gRPC.
                # ------------------------------------------------

                session.start()


            else:

                # =================================================
                # CASO 2:
                # LA LLAMADA YA EXISTE
                # =================================================
                #
                # No hacemos absolutamente nada.
                #
                # No asignamos otro VAD.
                #
                # El paquete continuará utilizando:
                #
                # call_id -> session existente
                #
                # Esto garantiza que una llamada permanezca
                # siempre en el mismo TEN VAD.
                # =================================================

                pass


        # --------------------------------------------------------
        # IMPORTANTE:
        #
        # El push se hace FUERA del lock.
        #
        # No queremos mantener el lock mientras manipulamos
        # la cola de audio.
        # --------------------------------------------------------

        session.push_audio(
            audio=audio,
            metadata_json=metadata_json,
        )


    # ========================================================
    # RELEASE
    # ========================================================

    def release(
        self,
        call_id,
    ):

        # --------------------------------------------------------
        # Protegemos el diccionario y pool.
        # --------------------------------------------------------

        with self.lock:

            # ------------------------------------------------
            # Sacamos la sesión.
            # ------------------------------------------------

            session = self.sessions.pop(
                call_id,
                None
            )


            # Si no existía, no hacemos nada.
            if session is None:
                return


            # ------------------------------------------------
            # El TEN VAD que estaba ocupado vuelve al pool.
            #
            # Ejemplo:
            #
            # call-001 estaba en VAD 12001
            #
            # al liberar:
            #
            # VAD 12001 vuelve a available
            # ------------------------------------------------

            self.available.append(
                (
                    session.vad_server,
                    session.out_port,
                )
            )


        # --------------------------------------------------------
        # Detenemos la sesión fuera del lock.
        #
        # stop() puede esperar al thread gRPC.
        # No queremos bloquear el pool durante ese tiempo.
        # --------------------------------------------------------

        session.stop()


        print(
            f"[MANAGER] Liberado "
            f"{call_id} → "
            f"{session.vad_server}"
        )


    # ========================================================
    # STOP ALL
    # ========================================================

    def stop_all(self):

        # --------------------------------------------------------
        # Copiamos las sesiones y limpiamos el diccionario.
        # --------------------------------------------------------

        with self.lock:

            sessions = list(
                self.sessions.items()
            )

            self.sessions.clear()


        # --------------------------------------------------------
        # Detenemos las sesiones una por una.
        # --------------------------------------------------------

        for call_id, session in sessions:

            print(
                f"[MANAGER] "
                f"Stopping {call_id}"
            )

            session.stop()

FRAME_MS = 20

running = True


def signal_handler(signum, frame):

    global running

    print("\n[HEADSET] Shutdown solicitado")

    running = False


def encode_udp_packet(
    call_id,
    metadata_json,
    audio,
):
    header = {
        "call_id": call_id,
        "metadata_json": metadata_json,
    }

    header_bytes = json.dumps(
        header,
        separators=(",", ":"),
    ).encode("utf-8")

    return (
        struct.pack(
            "!I",
            len(header_bytes),
        )
        + header_bytes
        + audio
    )


class UDPAudioSender:

    def __init__(
        self,
        host,
        port,
        copies,
        call_id_prefix,
    ):

        self.address = (
            host,
            port,
        )

        self.socket = socket.socket(
            socket.AF_INET,
            socket.SOCK_DGRAM,
        )

        self.targets = []

        for index in range(copies):

            call_id = (
                f"{call_id_prefix}-"
                f"{index + 1:03d}"
            )

            metadata_json = json.dumps(
                {
                    "cliente_id": (
                        f"12345-{index + 1}"
                    ),
                    "campana": "ventas",
                    "agente": (
                        f"agente-{index + 1:03d}"
                    ),

                    # El manager reemplazará
                    # estos valores con el
                    # OUT correspondiente.
                    "out_host": "",
                    "out_port": 0,
                },
                separators=(",", ":"),
            )

            self.targets.append(
                (
                    call_id,
                    metadata_json,
                )
            )

            print(
                f"[HEADSET] "
                f"{call_id} -> "
                f"UDP {host}:{port}"
            )

    def send(
        self,
        audio,
    ):

        for call_id, metadata_json in self.targets:

            packet = encode_udp_packet(
                call_id=call_id,
                metadata_json=metadata_json,
                audio=audio,
            )

            try:

                self.socket.sendto(
                    packet,
                    self.address,
                )

            except OSError as exc:

                print(
                    f"[HEADSET] ERROR "
                    f"{call_id}: {exc}",
                    file=sys.stderr,
                )

    def close(self):

        self.socket.close()


def main():

                )

                sender.send(
                    audio_bytes
                )

    except KeyboardInterrupt:

        pass

    except Exception as exc:

        print(
            f"[HEADSET] ERROR: {exc}",
            file=sys.stderr,
        )

        return 1

    finally:

from pathlib import Path
import threading

GENERATED_DIR = BASE_DIR / "generated"

    str(GENERATED_DIR)

import audio_listener_pb2_grpc


running = True


class SegmentOutputService(
    audio_listener_pb2_grpc.SegmentOutputServicer
):

    def __init__(
        self,
        output_directory,
    ):

        self.output_directory = Path(
            output_directory
        )

        self.output_directory.mkdir(
            parents=True,
            exist_ok=True,
        )

        self.lock = threading.Lock()

        print(
            f"[OUT] Audio directory: "
            f"{self.output_directory}"
        )

    def SendSegment(
        self,
        request,
        context,
    ):

        call_id = request.call_id
        segment_id = request.segment_id

        if not call_id:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "call_id es obligatorio"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        if not request.audio:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "audio es obligatorio"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        sample_rate = (
            request.sample_rate
            or 16000
        )

        channels = (
            request.channels
            or 1
        )

        if sample_rate != 16000:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "Solo se acepta audio a 16000 Hz"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        if channels != 1:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "Solo se acepta audio mono"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        audio_bytes = request.audio

        if len(audio_bytes) % 2 != 0:

            context.set_code(
                grpc.StatusCode.INVALID_ARGUMENT
            )

            context.set_details(
                "Audio PCM16 inválido"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        duration = (
            (len(audio_bytes) / 2)
            / sample_rate
        )

        call_directory = (
            self.output_directory /
            call_id
        )

        call_directory.mkdir(
            parents=True,
            exist_ok=True,
        )

        output_file = (
            call_directory /
            f"{segment_id}.wav"
        )

        try:

            with self.lock:

                with wave.open(
                    str(output_file),
                    "wb",
                ) as wav:

                    wav.setnchannels(
                        channels
                    )

                    wav.setsampwidth(
                        2
                    )

                    wav.setframerate(
                        sample_rate
                    )

                    wav.writeframes(
                        audio_bytes
                    )

        except Exception as exc:

            print(
                f"[OUT] ERROR guardando "
                f"{output_file}: {exc}",
                file=sys.stderr,
            )

            context.set_code(
                grpc.StatusCode.INTERNAL
            )

            context.set_details(
                "No se pudo guardar WAV"
            )

            return audio_listener_pb2.SegmentResponse(
                success=False
            )

        print(
            f"[OUT] "
            f"call={call_id} | "
            f"segment={segment_id} | "
            f"{duration:.3f}s | "
            f"{len(audio_bytes)} bytes | "
            f"{output_file}"
        )

        return audio_listener_pb2.SegmentResponse(
            success=True
        )

    def Ping(
        self,
        request,
        context,
    ):

        print(
            f"[OUT] Ping received | "
            f"source={request.source}"
        )

        return audio_listener_pb2.PingResponse(
            success=True,
            service="audio-out",
            message="OUT is alive",
        )
